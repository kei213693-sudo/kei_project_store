45static int diss_value_req_ind_handler(ke_msg_id_t const msgid,
46 struct diss_value_req_ind const *param,
47 ke_task_id_t const dest_id,
48 ke_task_id_t const src_id)
49{
50 // Initialize length
51 uint8_t len = 0;
52 // Pointer to the data
53 uint8_t *data = NULL;
54
55 // Check requested value
56 switch (param->value)
57 {
58 case DIS_MANUFACTURER_NAME_CHAR:
59 {
60 // Set information
61 len = APP_DIS_MANUFACTURER_NAME_LEN;
62 data = (uint8_t *)APP_DIS_MANUFACTURER_NAME;
63 } break;
64
65 case DIS_MODEL_NB_STR_CHAR:
66 {
67 // Set information
68 len = APP_DIS_MODEL_NB_STR_LEN;
69 data = (uint8_t *)APP_DIS_MODEL_NB_STR;
70 } break;
71
72 case DIS_SYSTEM_ID_CHAR:
73 {
74 // Set information
75 len = APP_DIS_SYSTEM_ID_LEN;
76 data = (uint8_t *)APP_DIS_SYSTEM_ID;
77 } break;
78
79 case DIS_PNP_ID_CHAR:
80 {
81 // Set information
82 len = APP_DIS_PNP_ID_LEN;
83 data = (uint8_t *)APP_DIS_PNP_ID;
84 } break;
85
86 case DIS_SERIAL_NB_STR_CHAR:
87 {
88 // Set information
89 len = APP_DIS_SERIAL_NB_STR_LEN;
90 data = (uint8_t *)APP_DIS_SERIAL_NB_STR;
91 } break;
92
93 case DIS_HARD_REV_STR_CHAR:
94 {
95 // Set information
96 len = APP_DIS_HARD_REV_STR_LEN;
97 data = (uint8_t *)APP_DIS_HARD_REV_STR;
98 } break;
99
100 case DIS_FIRM_REV_STR_CHAR:
101 {
102 // Set information
103 len = APP_DIS_FIRM_REV_STR_LEN;
104 data = (uint8_t *)APP_DIS_FIRM_REV_STR;
105 } break;
106
107 case DIS_SW_REV_STR_CHAR:
108 {
109 // Set information
110 len = APP_DIS_SW_REV_STR_LEN;
111 data = (uint8_t *)APP_DIS_SW_REV_STR;
112 } break;
113
114 case DIS_IEEE_CHAR:
115 {
116 // Set information
117 len = APP_DIS_IEEE_LEN;
118 data = (uint8_t *)APP_DIS_IEEE;
119 } break;
120
121 default:
122 ASSERT_ERR(0);
123 break;
124 }
125
126 // Allocate confirmation to send the value
127 struct diss_value_cfm *cfm_value = KE_MSG_ALLOC_DYN(DISS_VALUE_CFM,
128 src_id, dest_id,
129 diss_value_cfm,
130 len);
131
132 // Set parameters
133 cfm_value->value = param->value;
134 cfm_value->length = len;
135 if (len)
136 {
137 // Copy data
138 memcpy(&cfm_value->data[0], data, len);
139 }
140 // Send message
141 ke_msg_send(cfm_value);
142
143 return (KE_MSG_CONSUMED);
144}

156void app_dis_add_dis(void)
157{
158 struct diss_db_cfg* db_cfg;
159 // Allocate the DISS_CREATE_DB_REQ
160 struct gapm_profile_task_add_cmd *req = KE_MSG_ALLOC_DYN(GAPM_PROFILE_TASK_ADD_CMD,
161 TASK_GAPM, TASK_APP,
162 gapm_profile_task_add_cmd, sizeof(struct diss_db_cfg));
163 // Fill message
164 req->operation = GAPM_PROFILE_TASK_ADD;
165 #if (BLE_APP_AM0)
166 req->sec_lvl = PERM(SVC_AUTH, NO_AUTH);
167 #else
168 req->sec_lvl = PERM(SVC_AUTH, NO_AUTH);
169 #endif //(BLE_APP_AM0)
170 req->prf_task_id = TASK_ID_DISS;
171 req->app_task = TASK_APP;
172 req->start_hdl = 0;
173
174 // Set parameters
175 db_cfg = (struct diss_db_cfg* ) req->param;
176 db_cfg->features = APP_DIS_FEATURES;
177
178 // Send the message
179 ke_msg_send(req);
180}
