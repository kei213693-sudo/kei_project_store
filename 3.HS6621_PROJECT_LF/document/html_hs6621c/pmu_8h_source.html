957__STATIC_FORCEINLINE void pmu_cpu_reset(void)
958{
959 // flag
960 register_set1(&HS_PMU->SW_STATUS, PMU_SW_STATUS_REBOOT_FROM_SOFT_RESET_MASK);
961
962 // Use DMB to wait until all outstanding
963 // memory accesses are completed
964 __DMB();
965
966#if 0
967 register_set0(&HS_PMU->AHB_REMAP, PMU_CPU_SOFT_RESET_MASK);
968#else
969 // Read back PRIGROUP and merge with SYSRESETREQ
970 SCB->AIRCR = 0x05FA0001 | (SCB->AIRCR & 0x700);
971#endif
972}
