get the notification information：

一、Get the source of notification, handle the ANCSC_NOTIFICATION_IND event

// EventID Values
enum ancs_eventID_values
{
ANCS_EVENT_ID_NOTIFICATION_ADDED = 0,
ANCS_EVENT_ID_NOTIFICATION_MODIFIED = 1,
ANCS_EVENT_ID_NOTIFICATION_REMOVED = 2,
ANCS_EVENT_ID_RESERVED = 255,
};
// EventFlags
enum ancs_event_flags
{
ANCS_EVENT_FLAG_SILENT = (1 << 0),
ANCS_EVENT_FLAG_IMPORTANT = (1 << 1),
ANCS_EVENT_FLAG_EXISTING = (1 << 2),
ANCS_EVENT_FLAG_POSITIVEACTION = (1 << 3),
ANCS_EVENT_FLAG_NEGATIVEACTION = (1 << 4),
ANCS_EVENT_FLAG_RESERVED = 0,
};
// CategoryID Values
enum ancs_categoryID_values
{
ANCS_CATEGORY_ID_OTHER = 0,
ANCS_CATEGORY_ID_INCOMINGCALL = 1,
ANCS_CATEGORY_ID_MISSEDCALL = 2,
ANCS_CATEGORY_ID_VOICEMAIL = 3,
ANCS_CATEGORY_ID_SOCIAL = 4,
ANCS_CATEGORY_ID_SCHEDULE = 5,
ANCS_CATEGORY_ID_EMAIL = 6,
ANCS_CATEGORY_ID_NEWS = 7,
ANCS_CATEGORY_ID_HEALTHANDFITNESS = 8,
ANCS_CATEGORY_ID_BUSINESSANDFINANCE = 9,
ANCS_CATEGORY_ID_LOCATION = 10,
ANCS_CATEGORY_ID_ENTERTAINMENT = 11,
ANCS_CATEGORY_ID_RESERVED = 255,
};
//Parameters of the @ref ANCSC_NOTIFICATION_IND message
struct ancsc_notification_ind
{
uint8_t event_id; // ancs_eventID_values
uint8_t event_flags; // ancs_event_flags
uint8_t category_id; // ancs_categoryID_values
uint8_t category_count; // category count
uint32_t notification_uid; // message UID
};

二、Get the information

//Parameters of the @ref ANCSC_GET_NTF_ATTR_CMD message
struct ancsc_get_ntf_attr_cmd
{
uint32_t uid; // message UID
};
//Parameters of the @ref ANCSC_NTF_ATTR_IND message
struct ancsc_ntf_attr_ind
{
uint32_t notif_uid;
uint8_t date[ANCS_DATE_CAP];
uint16_t app_id_len;
uint16_t title_len;
uint16_t subtitle_len;
uint16_t msg_len;
uint8_t *app_id;
uint8_t *title;
uint8_t *subtitle;
uint8_t *msg;
uint8_t data[__ARRAY_EMPTY];
};
