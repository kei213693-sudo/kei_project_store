128static void app_send_indicate( void )
129{
130 sendSize = 0;
131 if( g_send_wechat.offset < g_send_wechat.len ){
132 if( g_send_wechat.len - g_send_wechat.offset > 20 ){
133 sendSize = 20;
134 }else{
135 sendSize = g_send_wechat.len - g_send_wechat.offset;
136 }
137 app_wechat_send_notify(&g_send_wechat.buf[g_send_wechat.offset], sendSize);
138 }else{
139 g_send_wechat.len = 0;
140 g_send_wechat.offset = 0;
141 memset(g_send_wechat.buf, 0, PKT_LENGTH);
142 }
143}

197static WeChat_Recv_Pkt g_recv_data = {NULL, 0, 0};

199static int app_wechat_write_req_handler(ke_msg_id_t const msgid,
200 struct wechat_write_ind *param,
201 ke_task_id_t const dest_id,
202 ke_task_id_t const src_id)
203{
204 int chunk_size = 0;
205 if(ke_state_get(TASK_APP) == APPM_CONNECTED){
206 if(param->length <= 20){
207 if( g_recv_data.len == 0 ){
208 BpFixHead *fix_head = (BpFixHead *)param->write_buf;
209 g_recv_data.len = ntohs( fix_head->nLength );
210 g_recv_data.offset = 0;
211 g_recv_data.buf = ke_malloc(g_recv_data.len, KE_MEM_KE_MSG);
212 }
213 chunk_size = g_recv_data.len - g_recv_data.offset;
214 chunk_size = chunk_size < param->length ? chunk_size : param->length;
215 memcpy( g_recv_data.buf + g_recv_data.offset, param->write_buf, chunk_size );
216 g_recv_data.offset += chunk_size;
217 if( g_recv_data.len <= g_recv_data.offset ){
218 m_wechat_handler->m_data_consume_func( g_recv_data.buf, g_recv_data.len );
219 ke_free(g_recv_data.buf);
220 g_recv_data.buf = NULL;
221 g_recv_data.len = 0;
222 g_recv_data.offset = 0;
223 }
224 }
225 }
226
227 return (KE_MSG_CONSUMED);
228}
