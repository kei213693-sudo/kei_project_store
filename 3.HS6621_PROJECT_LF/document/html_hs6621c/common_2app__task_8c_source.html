75static uint8_t app_get_handler(const struct app_subtask_handlers *handler_list_desc,
76 ke_msg_id_t msgid,
77 void *param,
78 ke_task_id_t src_id)
79{
80 // Counter
81 uint8_t counter;
82
83 // Get the message handler function by parsing the message table
84 for (counter = handler_list_desc->msg_cnt; 0 < counter; counter--)
85 {
86 struct ke_msg_handler *handler
87 = (struct ke_msg_handler*)(handler_list_desc->p_msg_handler_tab + counter - 1);
88
89 if ((handler->id == msgid) ||
90 (handler->id == KE_MSG_DEFAULT_HANDLER))
91 {
92 // If handler is NULL, message should not have been received in this state
93 ASSERT_ERR(handler->func);
94
95 return (uint8_t)(handler->func(msgid, param, TASK_APP, src_id));
96 }
97 }
98
99 // If we are here no handler has been found, drop the message
100 return (KE_MSG_CONSUMED);
101}

248static int gapm_cmp_evt_handler(ke_msg_id_t const msgid,
249 struct gapm_cmp_evt const *param,
250 ke_task_id_t const dest_id,
251 ke_task_id_t const src_id)
252{
253 #if ((NVDS_SUPPORT) && (BLE_APP_SEC))
254 nvds_tag_len_t key_len = KEY_LEN;
255 #endif //(NVDS_SUPPORT)
256 switch(param->operation)
257 {
258 // Reset completed
259 case (GAPM_RESET):
260 {
261 if(param->status == GAP_ERR_NO_ERROR)
262 {
263 #if (BLE_APP_HID)
264 //app_hid_start_mouse();
265 #endif //(BLE_APP_HID)
266 // Set Device configuration
267 struct gapm_set_dev_config_cmd* cmd = KE_MSG_ALLOC(GAPM_SET_DEV_CONFIG_CMD,
268 TASK_GAPM, TASK_APP,
269 gapm_set_dev_config_cmd);
270 // Set the operation
271 cmd->operation = GAPM_SET_DEV_CONFIG;
272 // Set the device role - Peripheral
273 cmd->role = GAP_ROLE_PERIPHERAL;
274 // Do not support secure connections
275 cmd->pairing_mode = GAPM_PAIRING_LEGACY;
276 // Set Data length parameters
277 cmd->sugg_max_tx_octets = BLE_MAX_OCTETS;
278 cmd->sugg_max_tx_time = BLE_MAX_TIME;
279 //cmd->max_mtu = 200;
281 cmd->tx_pref_phy = GAP_PHY_LE_2MBPS;
283 cmd->rx_pref_phy = GAP_PHY_LE_2MBPS;
284 // Maximal MTU/MPS acceptable for device
285 cmd->max_mtu = 251;
286 cmd->max_mps = GAP_MAX_LE_MTU;
287
288 #if (BLE_APP_HID)
289 // Enable Slave Preferred Connection Parameters present
290 cmd->att_cfg = GAPM_MASK_ATT_SLV_PREF_CON_PAR_EN;
291 #endif //(BLE_APP_HID)
292
293 // Host privacy disabled by default
294 cmd->privacy_cfg = GAPM_PRIV_CFG_PRIV_ADDR_BIT;
295 memcpy(cmd->addr.addr, &app_env.addr.addr[0], 6);
296
297 #if ((NVDS_SUPPORT) && (BLE_APP_SEC))
298
299 if ((app_sec_get_bond_status()==true) &&
300 (nvds_get(NVDS_TAG_LOC_IRK, &key_len, app_env.loc_irk) == NVDS_OK))
301 {
302 memcpy(cmd->irk.key, app_env.loc_irk, 16);
303 }
304 else
305 #endif //(NVDS_SUPPORT)
306 {
307 memset((void *)&cmd->irk.key[0], 0x00, KEY_LEN);
308 }
309 #if (BLE_APP_PRIVACY)
310 cmd->privacy_cfg |= GAPM_PRIV_CFG_PRIV_EN_BIT;
311 cmd->renew_dur = 0x0384;
312 #endif
313 // Send message
314 ke_msg_send(cmd);
315 }
316 else
317 {
318 ASSERT_ERR(0);
319 }
320 }
321 break;
322
323 // Device Configuration updated
324 #if (BLE_APP_WHITE_LIST || BLE_APP_PRIVACY)
325 case (GAPM_SET_DEV_CONFIG):
326 {
327 #if (BLE_APP_WHITE_LIST)
328 appm_whl_cmp_evt_handler(msgid, param, dest_id, src_id);
329 #endif
330 #if (BLE_APP_PRIVACY)
331 appm_privacy_cmp_evt_handler(msgid, param, dest_id, src_id);
332 #endif
333 #if(BLE_APP_SEC)
334 app_gap_set_loc_irk();
335 #endif
336 }
337 break;
338 #else
339 case (GAPM_SET_DEV_CONFIG):
340 {
341 ASSERT_INFO(param->status == GAP_ERR_NO_ERROR, param->operation, param->status);
342
343 #if(BLE_APP_SEC)
344 app_gap_set_loc_irk();
345 #endif
346
347 appm_reg_svc_itf();
348 // Add the first required service in the database
349 // and wait for the PROFILE_ADDED_IND
350 appm_add_svc();
351 }
352 break;
353 #endif
354 #if (BLE_APP_WHITE_LIST)
355 case (GAPM_GET_WLIST_SIZE):
356 case (GAPM_SET_WL):
357 {
358 if(appm_whl_cmp_evt_handler(msgid, param, dest_id, src_id) != KE_MSG_CONSUMED)
359 {
360 ke_state_t state = ke_state_get(dest_id);
361
362 if (state < APPM_READY) {
363 appm_reg_svc_itf();
364 appm_add_svc();
365 }
366 }
367 }
368 break;
369 #endif
370 case (GAPM_PROFILE_TASK_ADD):
371 {
372 //log_debug("GAPM_PROFILE_TASK_ADD, st:%d\n", param->status);
373 // Add the next requested service
374 if (!appm_add_svc())
375 {
376 // Go to the ready state
377 ke_state_set(TASK_APP, APPM_READY);
378 // No more service to add, start advertising
379 //appm_adv_update_state(true);
380 #if (APP_CUSTOMER_ID == 1)
381 extern void appm_pair_advdata_set(void);
382 appm_pair_advdata_set();
383 #endif
384 appm_adv_start();
385 }
386 }
387 break;
388
389 case (GAPM_GEN_RAND_NB) :
390 {
391 if (app_env.rand_cnt == 1)
392 {
393 // Generate a second random number
394 app_env.rand_cnt++;
395 struct gapm_gen_rand_nb_cmd *cmd = KE_MSG_ALLOC(GAPM_GEN_RAND_NB_CMD,
396 TASK_GAPM, TASK_APP,
397 gapm_gen_rand_nb_cmd);
398 cmd->operation = GAPM_GEN_RAND_NB;
399 ke_msg_send(cmd);
400 }
401 else
402 {
403 struct gapm_set_irk_cmd *cmd = KE_MSG_ALLOC(GAPM_SET_IRK_CMD,
404 TASK_GAPM, TASK_APP,
405 gapm_set_irk_cmd);
406 app_env.rand_cnt=0;
408 cmd->operation = GAPM_SET_IRK;
409 memcpy(&cmd->irk.key[0], &app_env.loc_irk[0], KEY_LEN);
410 ke_msg_send(cmd);
411 }
412 }
413 break;
414
415 case (GAPM_SET_IRK):
416 {
417 // ASSERT_INFO(param->status == GAP_ERR_NO_ERROR, param->operation, param->status);
418
419 // If not Bonded already store the generated value in NVDS
420 #if(BLE_APP_SEC)
421 if (app_sec_get_bond_status()==false)
422 {
423 #if (NVDS_SUPPORT)
424 if (nvds_put(NVDS_TAG_LOC_IRK, KEY_LEN, (uint8_t *)&app_env.loc_irk) != NVDS_OK)
425 #endif //(NVDS_SUPPORT)
426 {
427 ASSERT_INFO(0, 0, 0);
428 }
429 }
430 #endif
431
432 app_env.rand_cnt = 0;
433 }
434 break;
435
436 #if (BLE_APP_PRIVACY)
437 case (GAPM_GET_RAL_SIZE):
438 case (GAPM_SET_RAL):
439 appm_privacy_cmp_evt_handler(msgid, param, dest_id, src_id);
440 break;
441 #endif
442
443 case (GAPM_CREATE_ADV_ACTIVITY):
444 case (GAPM_STOP_ACTIVITY):
445 case (GAPM_START_ACTIVITY):
446 case (GAPM_DELETE_ACTIVITY):
447 case (GAPM_SET_ADV_DATA):
448 case (GAPM_SET_SCAN_RSP_DATA):
449 case (GAPM_DELETE_ALL_ACTIVITIES) :
450 {
451 appm_adv_cmp_evt_handler((void*)param);
452 } break;
453
454 default:
455 {
456 // Drop the message
457 }
458 break;
459 }
460
461 return (KE_MSG_CONSUMED);
462}

564static int gapc_connection_req_ind_handler(ke_msg_id_t const msgid,
565 struct gapc_connection_req_ind const *param,
566 ke_task_id_t const dest_id,
567 ke_task_id_t const src_id)
568{
569 app_env.conidx = KE_IDX_GET(src_id);
570 #if(BLE_APP_SEC)
571 bool is_bond = app_sec_get_bond_status_by_addr(param->peer_addr);
572 #endif
573
574 //log_debug("Device type(%d) conidx(%d) connected, ", param->peer_addr_type, app_env.conidx);
575 //log_debug_array_ex("ADDR", &param->peer_addr, 6);
576 // Check if the received Connection Handle was valid
577 if (app_env.conidx != GAP_INVALID_CONIDX)
578 {
579 // Retrieve the connection info from the parameters
580 app_env.conhdl = param->conhdl;
581
582 // Send connection confirmation
583 struct gapc_connection_cfm *cfm = KE_MSG_ALLOC(GAPC_CONNECTION_CFM,
584 KE_BUILD_ID(TASK_GAPC, app_env.conidx), TASK_APP,
585 gapc_connection_cfm);
586
587 #if(BLE_APP_SEC)
588 cfm->auth = is_bond ? GAP_AUTH_REQ_MITM_BOND : GAP_AUTH_REQ_MITM_BOND; // TODO [FBE] restore valid data
589 #else // !(BLE_APP_SEC)
590 cfm->auth = GAP_AUTH_REQ_NO_MITM_NO_BOND;
591 #endif // (BLE_APP_SEC)
592 // Send the message
593 ke_msg_send(cfm);
594
595 /*--------------------------------------------------------------
596 * ENABLE REQUIRED PROFILES
597 *--------------------------------------------------------------*/
598
599 #if (BLE_APP_BATT)
600 // Enable Battery Service
601 app_batt_enable_prf(app_env.conhdl);
602 #endif //(BLE_APP_BATT)
603
604 #if (BLE_APP_HID)
605 // Enable HID Service
606 app_hid_enable_prf(app_env.conidx);
607 #endif //(BLE_APP_HID)
608
609 #if (BLE_APP_SIMPLE_SERVER)
610 // Enable SIMPLE_SERVER Service
611 app_simple_server_enable_prf(app_env.conhdl);
612 #endif //(BLE_APP_SIMPLE_SERVER)
613
614 #if (BLE_APP_WECHAT)
615 //TODO: bd_addr_t mac_addr should be defined
616 app_wechat_enable_prf(app_env.conhdl, &app_env.addr);
617 #endif //(BLE_APP_WECHAT)
618
619 #if (BLE_TSPP_SERVER)
620 // Enable TSPP_SERVER Service
621 app_tspp_server_enable_prf(app_env.conidx);
622 #endif //(BLE_APP_TSPP_SERVER)
623
624 #if (BLE_ATVV_SERVER)
625 // Enable TSPP_SERVER Service
626 app_atvv_server_enable_prf(app_env.conidx);
627 #endif //(BLE_APP_TSPP_SERVER)
628
629 #if (BLE_APP_ANCSC)
630 app_ancsc_enable_prf(app_env.conidx);
631 #endif //(BLE_APP_ANCSC)
632
633 // We are now in connected State
634 ke_state_set(dest_id, APPM_CONNECTED);
635
636
637 #if (BLE_APP_SEC)
638 if (is_bond)
639 {
640 // Ask for the peer device to either start encryption
641 //app_sec_send_security_req(app_env.conidx);
642 }
643 #endif // (BLE_APP_SEC)
644 APP_HANDLER(gap_connected, msgid, param, dest_id, src_id); // APP layer Callback
645 }
646 else
647 {
648 #if (APP_CUSTOMER_ID == 1)
649 extern void appm_pair_advdata_set(void);
650 appm_pair_advdata_set();
651 #endif
652 // No connection has been established, restart advertising
653 appm_adv_update_state(true);
654 }
655
656 return (KE_MSG_CONSUMED);
657}

806static int appm_msg_handler(ke_msg_id_t const msgid,
807 void *param,
808 ke_task_id_t const dest_id,
809 ke_task_id_t const src_id)
810{
811 // Retrieve identifier of the task from received message
812 ke_task_id_t src_task_id = MSG_T(msgid);
813 // Message policy
814 uint8_t msg_pol = KE_MSG_CONSUMED;
815
816 switch (src_task_id)
817 {
818 case (TASK_ID_GAPC):
819 {
820 #if (BLE_APP_SEC)
821 if ((msgid >= GAPC_BOND_CMD) &&
822 (msgid <= GAPC_SECURITY_IND))
823 {
824 // Call the Security Module
825 msg_pol = app_get_handler(&app_sec_handlers, msgid, param, src_id);
826 }
827 #endif //(BLE_APP_SEC)
828 // else drop the message
829 } break;
830
831 case (TASK_ID_GATTC):
832 {
833 // Service Changed - Drop
834 } break;
835
836 #if (BLE_APP_DIS)
837 case (TASK_ID_DISS):
838 {
839 // Call the Device Information Module
840 msg_pol = app_get_handler(&app_dis_handlers, msgid, param, src_id);
841 } break;
842 #endif //(BLE_APP_DIS)
843
844 #if (BLE_APP_SIMPLE_SERVER)
845 case (TASK_ID_SIMPLE_SERVER):
846 {
847 // Call the Audio Mode 0 Module
848 msg_pol = app_get_handler(&app_simple_server_handlers, msgid, param, src_id);
849 } break;
