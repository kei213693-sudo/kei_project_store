92uint8_t appm_whl_set_list(void)
93{
94 uint8_t bond_num = 0; // when nvds modify, the bond num get from nvds
95 uint8_t nvds_len = NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST;
96 // BD Address of the peer device
97 struct gap_bdaddr peer_bd_addr[NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST];
98
99 if (app_whl_env.enable == 0)
100 {
101 return 0;
102 }
103
104 // get from nvds
105 if (app_sec_get_bond_status())
106 {
107 struct app_sec_conn_info_tag info;
108 nvds_tag_len_t length = sizeof(struct app_sec_conn_info_tag);
109 uint8_t index = 0;
110
111 // Get bond status from NVDS
112 for(index = 0; index <=nvds_len; index++)
113 {
114 if (nvds_get(NVDS_TAG_BLE_LINK_KEY_LAST-index, &length, (uint8_t *)&info) == NVDS_OK)
115 {
116 #if (BLE_APP_PRIVACY)
117 memcpy(&peer_bd_addr[bond_num], &info.irk.addr, sizeof(struct gap_bdaddr));
118 #else
119 memcpy(&peer_bd_addr[bond_num], &info.id_addr, sizeof(struct gap_bdaddr));
120 #endif
121 bond_num++;
122 }
123 }
124 }
125
126 if (bond_num == 0) return 0;
127
128 struct gapm_list_set_wl_cmd* cmd = KE_MSG_ALLOC_DYN(GAPM_LIST_SET_CMD,
129 TASK_GAPM, TASK_APP,
130 gapm_list_set_wl_cmd, sizeof(struct gap_bdaddr)*bond_num);
131 cmd->operation = GAPM_SET_WL;
132
133 cmd->size = bond_num;
134 for(uint8_t i=0;i<bond_num;i++)
135 {
136 memcpy(&cmd->wl_info[i], &peer_bd_addr[i], sizeof(struct gap_bdaddr));
137 }
138 ke_msg_send(cmd);
139 return bond_num;
140}
