81void app_batt_add_bas(void)
82{
83 struct bass_db_cfg* db_cfg;
84 // Allocate the BASS_CREATE_DB_REQ
85 struct gapm_profile_task_add_cmd *req = KE_MSG_ALLOC_DYN(GAPM_PROFILE_TASK_ADD_CMD,
86 TASK_GAPM, TASK_APP,
87 gapm_profile_task_add_cmd, sizeof(struct bass_db_cfg));
88 // Fill message
89 req->operation = GAPM_PROFILE_TASK_ADD;
90 req->sec_lvl = PERM(SVC_AUTH, NO_AUTH);
91 req->prf_task_id = TASK_ID_BASS;
92 req->app_task = TASK_APP;
93 req->start_hdl = 0;
94
95 // Set parameters
96 db_cfg = (struct bass_db_cfg* ) req->param;
97
98 // Add a BAS instance
99 db_cfg->bas_nb = 1;
100 // Sending of notifications is supported
101 db_cfg->features[0] = BAS_BATT_LVL_NTF_SUP;
102
103 // Send the message
104 ke_msg_send(req);
105}
