178static void appm_build_adv_data(uint16_t max_length, uint16_t *p_length, uint8_t *p_buf)
179{
180 // Set name
181 {
182 // Device name length
183 *p_buf = app_env.dev_name_len + 1;
184 // Device name flag (check if device name is complete or not)
185 *(p_buf + 1) = '\x09';
186 // Copy device name
187 memcpy(p_buf + 2, app_env.dev_name, app_env.dev_name_len);
188 // Update advertising data length
189 *p_length += (app_env.dev_name_len + 2);
190 p_buf += *p_length;
191 }
192
193 #if 0
194 // Set appearance
195 memcpy(p_buf, APP_ADV_DATA_APPEARANCE, APP_ADV_DATA_APPEARANCE_LEN);
196 *p_length += APP_ADV_DATA_APPEARANCE_LEN;
197 p_buf += *p_length;
198 #endif
199
200 // Set list of UUIDs
201 memcpy(p_buf, APP_ADV_DATA_MANU, APP_ADV_DATA_MANU_LEN);
202 memcpy(&p_buf[4], &app_env.addr.addr[0], 6);
203 for(int i=0; i<3; ++i)
204 {
205 char tmp = p_buf[4+i];
206 p_buf[4+i] = p_buf[4+5-i];
207 p_buf[4+5-i] = tmp;
208 }
209 *p_length += APP_ADV_DATA_MANU_LEN;
210 p_buf += *p_length;
211
212 // Sanity check
213 ASSERT_ERR(rem_len >= max_length);
214}

259void appm_init(void)
260{
261 // Reset the application manager environment
262 memset(&app_env, 0, sizeof(app_env));
263
264 // Create APP task
265 ke_task_create(TASK_APP, &TASK_DESC_APP);
266
267 // Initialize Task state
268 ke_state_set(TASK_APP, APPM_INIT);
269
270 // Get the Device Name to add in the Advertising Data (Default one or NVDS one)
271 #if (NVDS_SUPPORT)
272 app_env.dev_name_len = APP_DEVICE_NAME_MAX_LEN;
273 if (nvds_get(NVDS_TAG_DEVICE_NAME, &(app_env.dev_name_len), app_env.dev_name) != NVDS_OK)
274 #endif //(NVDS_SUPPORT)
275 {
276 // Get default Device Name (No name if not enough space)
277 memcpy(app_env.dev_name, APP_DFLT_DEVICE_NAME, APP_DFLT_DEVICE_NAME_LEN);
278 app_env.dev_name_len = APP_DFLT_DEVICE_NAME_LEN;
279
280 // TODO update this value per profiles
281 }
282
283 #if (NVDS_SUPPORT)
284 nvds_tag_len_t len=6;
285 if (nvds_get(NVDS_TAG_BD_ADDRESS, &len, &app_env.addr.addr[0]) == NVDS_OK)
286 {
287 }
288 else
289 #endif //(NVDS_SUPPORT)
290 {
291 memcpy(&app_env.addr.addr[0], &co_default_bdaddr,sizeof(co_default_bdaddr));
292 }
293 #if defined(DEBUG_ADDR)
294 memcpy(&app_env.addr.addr[0], DEBUG_ADDR, 6); // !! JUST FOR DEBUG !!
295 #endif
296
297 /*------------------------------------------------------
298 * INITIALIZE ALL MODULES
299 *------------------------------------------------------*/
300
301 // load device information:
302
303 #if (BLE_APP_SEC)
304 // Security Module
305 app_sec_init();
306 #endif // (BLE_APP_SEC)
307
308 #if (BLE_APP_DIS)
309 // Device Information Module
310 app_dis_init();
311 #endif //(BLE_APP_DIS)
312
313 #if (BLE_APP_SIMPLE_SERVER)
314 // simple_server Module
315 app_simple_server_init();
316 #endif //(BLE_APP_SIMPLE_SERVER)
317
318 #if (BLE_APP_WECHAT)
319 // Device Information Module
320 app_wechat_init();
321 #endif //(BLE_APP_WECHAT)
322
323 #if (BLE_APP_ANCSC)
324 // ANCS Module
325 app_ancsc_init();
326 #endif //(BLE_APP_ANCSC)
327 #if (BLE_APP_HID)
328 app_hid_init();
329 #endif //(BLE_APP_HID)
330 // Reset the stack
331 appm_send_gapm_reset_cmd();
332
333 // start adversting
334 appm_advertising_init();
335}

365bool appm_add_svc(void)
366{
367 // Indicate if more services need to be added in the database
368 bool more_svc = false;
369
370 // Check if another should be added in the database
371 if (app_env.next_svc != APPM_SVC_LIST_STOP)
372 {
373 ASSERT_INFO(appm_add_svc_func_list[app_env.next_svc] != NULL, app_env.next_svc, 1);
374
375 // Call the function used to add the required service
376 appm_add_svc_func_list[app_env.next_svc]();
377
378 // Select following service to add
379 app_env.next_svc++;
380 more_svc = true;
381
382 // Go to the create db state
383 ke_state_set(TASK_APP, APPM_CREATE_DB);
384 }
385
386 return more_svc;
387}
