101void app_atvv_recv_handler(struct atvv_server_write_ind const *param)
102{
103 uint8_t data[32];
104
105 memcpy(data, param->value, param->length);
106 log_debug_array_ex("[ATVV] atvv received data", data, param->length);
107
108 switch(data[0])
109 {
110 case CMD_ATVV_GET_CAPS:
111 log_debug("[ATVV] CMD_ATVV_GET_CAPS\r\n");
112 _atvv_get_caps_respose(param->conidx);
113 break;
114
115 case CMD_ATVV_MIC_OPEN:
116 log_debug("[ATVV] CMD_ATVV_MIC_OPEN\r\n");
117 _atvv_mic_open(param->conidx);
118 break;
119
120 case CMD_ATVV_MIC_CLOSE:
121 log_debug("[ATVV] CMD_ATVV_MIC_CLOSE\r\n");
122 _atvv_mic_close(param->conidx);
123 break;
124
125 case CMD_ATVV_SEARCH_END:
126 log_debug("[ATVV] CMD_ATVV_SEARCH_END\r\n");
127 _atvv_search_end();
128 break;
129
130 default:
131 log_debug("[ATVV] unkown command\r\n");
132 break;
133 }
134}

136void app_atvv_notify_audio_frame(uint8_t *input, uint8_t len, int16_t prevsample, uint16_t previndex)
137{
138 uint8_t buf[200];
139
140 buf[0] = app_atvv_server_env.seq_num >> 8;
141 buf[1] = app_atvv_server_env.seq_num & 0x00FF;
142 buf[2] = 0x01;
143 buf[3] = (uint8_t)(prevsample >> 8);
144 buf[4] = prevsample & 0x00FF;
145 buf[5] = previndex & 0x00FF;
146
147 memcpy(buf+6, input, len);
148 app_atvv_server_send_data(0, buf, len+6);
149
150 app_atvv_server_env.seq_num++;
151}

158static void _atvv_get_caps_respose(uint8_t conidx)
159{
160 uint8_t caps[] = {
161 CTL_GET_CAPS_RESP,
162 0x00, 0x04, // VER:0.4
163 0x00, 0x01, // CODEC: ADPCM/8k/16 & 16k/16)
164 0x00, 0x86, // 134 bytes/frame
165 0x00, 0x14 // 20 bytes/characteristic
166 };
167 app_atvv_server_send_control(conidx, &caps[0], 9);
168
169 log_debug_array_ex("[ATVV] Get capability response ", caps, 9);
170}

177static void _atvv_mic_open(uint8_t conidx)
178{
179 uint8_t buf = CTL_AUDIO_START;
180 app_atvv_server_send_control(conidx, &buf, sizeof(buf));
181 co_timer_set(&audio_delay_stop_timer, 30000, TIMER_ONE_SHOT, audio_delay_stop_timer_handler, NULL);
182
183 app_atvv_server_env.seq_num = 0;
184 app_audio_start();
185}

189 app_audio_stop();

192 app_atvv_server_send_control(conidx, &buf, 1);
193}
