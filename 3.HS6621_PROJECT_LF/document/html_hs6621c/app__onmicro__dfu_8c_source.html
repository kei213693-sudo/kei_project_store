93void app_onmicro_dfu_update_end_ind_handler(uint8_t status, void *p)
94{
95 log_debug("onmicro_dfu_update_end. status: %d\n", status);
96 if(status == DFU_UPDATE_ST_SUCCESS)
97 {
98#if CONFIG_HS6621C_VROM
99 do {
100 uint16_t ctrl_flag = *(uint16_t*)p;
101 if(ctrl_flag & DFU_CTRL_BIT_MORE_IMG){
102 struct vrom_dfu_info {
103 uint8_t bdaddr[6];
104 uint8_t addr_type;
105 int8_t load_capacitance;
106 uint16_t dfu_att_handle;
107 };
108 extern int vrom_set_dfu_info(struct vrom_dfu_info* p_dfu_info);
109 struct vrom_dfu_info dfu_info;
110 memcpy(dfu_info.bdaddr, app_env.bdaddr, sizeof(app_env.bdaddr));
111 dfu_info.addr_type = app_env.addr_type;
112 dfu_info.dfu_att_handle = app_env.dfu_att_handle;
113 dfu_info.load_capacitance = app_env.load_capacitance;
114 int res = vrom_set_dfu_info(&dfu_info);
115 co_assert(res >= 0);
116 }
117 } while(0);
118#endif
119 ke_state_t state = ke_state_get(TASK_APP);
120 if(state != APPM_CONNECTED){
121 log_debug("REBOOT !!\n");
122 pmu_force_reboot();
123 }else{
124 log_debug("Disconnecting.\n");
125 appm_disconnect(0);
126 onmicro_dfu_reset_flag = true;
127 }
128 }
129 else
130 {
131 struct gapc_conn_param conn_param = {0x20, 0x40, 10, 200};
132 appm_update_param(&conn_param);
133 }
134}
