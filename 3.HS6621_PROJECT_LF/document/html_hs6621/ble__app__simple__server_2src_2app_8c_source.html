133 static void appm_build_adv_data(uint16_t max_length, uint16_t *p_length, uint8_t *p_buf)
134 {
135  // Remaining Length
136  uint8_t rem_len = max_length;
137  // Set appearance
138  memcpy(p_buf, APP_ADV_DATA_APPEARANCE, APP_ADV_DATA_APPEARANCE_LEN);
139  *p_length += APP_ADV_DATA_APPEARANCE_LEN;
140  p_buf += APP_ADV_DATA_APPEARANCE_LEN;
141 
142  // Set list of UUIDs
143  memcpy(p_buf, APP_ADV_DATA_UUID, APP_ADV_DATA_UUID_LEN);
144  *p_length += APP_ADV_DATA_UUID_LEN;
145  p_buf += APP_ADV_DATA_UUID_LEN;
146 
147  // Sanity check
148  ASSERT_ERR(rem_len >= max_length);
149 
150  // Get remaining space in the Advertising Data - 2 bytes are used for name length/flag
151  rem_len -= *p_length;
152 
153  // Check if additional data can be added to the Advertising data - 2 bytes needed for type and length
154  if (rem_len > 2)
155  {
156  uint8_t dev_name_length = co_min(app_env.dev_name_len, (rem_len - 2));
157 
158  // Device name length
159  *p_buf = dev_name_length + 1;
160  // Device name flag (check if device name is complete or not)
161  *(p_buf + 1) = (dev_name_length == app_env.dev_name_len) ? '\x08' : '\x09';
162  // Copy device name
163  memcpy(p_buf + 2, app_env.dev_name, dev_name_length);
164 
165  // Update advertising data length
166  *p_length += (dev_name_length + 2);
167  }
168 }

214 void appm_init(void)
215 {
216  // Reset the application manager environment
217  memset(&app_env, 0, sizeof(app_env));
218 
219  // Create APP task
220  ke_task_create(TASK_APP, &TASK_DESC_APP);
221 
222  // Initialize Task state
223  ke_state_set(TASK_APP, APPM_INIT);
224 
225  // Get the Device Name to add in the Advertising Data (Default one or NVDS one)
226  #if (NVDS_SUPPORT)
227  app_env.dev_name_len = APP_DEVICE_NAME_MAX_LEN;
228  if (nvds_get(NVDS_TAG_DEVICE_NAME, &(app_env.dev_name_len), app_env.dev_name) != NVDS_OK)
229  #endif //(NVDS_SUPPORT)
230  {
231  // Get default Device Name (No name if not enough space)
232  memcpy(app_env.dev_name, APP_DFLT_DEVICE_NAME, APP_DFLT_DEVICE_NAME_LEN);
233  app_env.dev_name_len = APP_DFLT_DEVICE_NAME_LEN;
234 
235  // TODO update this value per profiles
236  }
237 
238 
239  /*------------------------------------------------------
240  * INITIALIZE ALL MODULES
241  *------------------------------------------------------*/
242 
243  // load device information:
244 
245  #if (BLE_APP_SEC)
246  // Security Module
247  app_sec_init();
248  #endif // (BLE_APP_SEC)
249 
250  #if (BLE_APP_DIS)
251  // Device Information Module
252  app_dis_init();
253  #endif //(BLE_APP_DIS)
254 
255  #if (BLE_APP_SIMPLE_SERVER)
256  // simple_server Module
257  app_simple_server_init();
258  #endif //(BLE_APP_SIMPLE_SERVER)
259 
260  // Reset the stack
261  appm_send_gapm_reset_cmd();
262 
263  // start adversting
264  appm_advertising_init();
265 }

267 bool appm_add_svc(void)
268 {
269  // Indicate if more services need to be added in the database
270  bool more_svc = false;
271 
272  // Check if another should be added in the database
273  if (app_env.next_svc != APPM_SVC_LIST_STOP)
274  {
275  ASSERT_INFO(appm_add_svc_func_list[app_env.next_svc] != NULL, app_env.next_svc, 1);
276 
277  // Call the function used to add the required service
278  appm_add_svc_func_list[app_env.next_svc]();
279 
280  // Select following service to add
281  app_env.next_svc++;
282  more_svc = true;
283 
284  // Go to the create db state
285  ke_state_set(TASK_APP, APPM_CREATE_DB);
286  }
287 
288  return more_svc;
289 }
