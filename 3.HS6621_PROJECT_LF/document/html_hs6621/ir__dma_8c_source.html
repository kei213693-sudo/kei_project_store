122 static HS_DMA_CH_Type* ir_tim_dma_config(HS_TIM_Type *tim, HS_DMA_CH_Type *dma, const ir_tim_dma_config_t*config)
123 {
124  dma_id_t dma_id;
125 
126  if(dma == NULL)
127  dma = dma_allocate();
128 
129  dma_id = tim==HS_TIM0 ? TIMER0_DMA_ID : (tim==HS_TIM1 ? TIMER1_DMA_ID : TIMER2_DMA_ID);
130 
131  if(dma)
132  {
133  bool res;
134  dma_config_t dconfig;
135 
136  dconfig.slave_id = dma_id;
137  dconfig.direction = DMA_MEM_TO_DEV;
138  dconfig.src_addr_width = DMA_SLAVE_BUSWIDTH_16BITS;
139  dconfig.dst_addr_width = DMA_SLAVE_BUSWIDTH_16BITS;
140  dconfig.src_burst = DMA_BURST_LEN_1UNITS;
141  dconfig.dst_burst = DMA_BURST_LEN_1UNITS;
142  dconfig.dev_flow_ctrl = false;
143  dconfig.priority = 0;
144  dconfig.callback = config->callback;
145 
146  dconfig.lli.enable = false;
147  dconfig.lli.use_fifo = config->use_fifo;
148  dconfig.lli.src_addr = (uint32_t)config->buffer;
149  dconfig.lli.dst_addr = (uint32_t)&tim->DMAR;
150  dconfig.lli.block_num = config->block_num;
151  dconfig.lli.block_len = config->buffer_len/config->block_num;
152  dconfig.lli.llip = config->block_llip;
153 
154  res = dma_config(dma, &dconfig);
155  if(!res)
156  {
157  dma_release(dma);
158  return NULL;
159  }
160 
161  tim->DCR = ((config->num - 1)<<8) | ((uint32_t)config->tim_addr-(uint32_t)tim)/4;
162  tim->DIER |= config->req;
163  }
164 
165  return dma;
166 }

327 void ir_dma_stop(void)
328 {
329  if(!ir_send_flag)
330  return;//ir is already stopped
331 
332  if(ir_repeat_count < 2)
333  {
334  ir_stop_dealy_flag = true;
335  return;//must send one ir repeat frame at least
336  }
337 
338 // log_debug("%s @ %d\n", __FUNCTION__, __LINE__);
339 
340  tim_stop(HS_TIM0);
341  dma_stop(tim_dma);
342  dma_release(tim_dma);
343  tim_dma = NULL;
344 
345  co_timer_del(&ir_repeat_timer);
346  ir_send_flag = false;
347 }
