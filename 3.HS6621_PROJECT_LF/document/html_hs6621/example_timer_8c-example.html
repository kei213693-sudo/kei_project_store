This is an example of how to use the timer

/*********************************************************************
* INCLUDES
*/
#include "peripheral.h"
/*********************************************************************
* LOCAL FUNCTIONS
*/
void tim_timer_handler(void)
{
log_debug("tick\n");
}
/*********************************************************************
* PUBLIC FUNCTIONS
*/
void example_timer(void)
{
const tim_config_t timer_cfg =
{
.mode = TIM_TIMER_MODE,
.config.timer =
{
.period_us = 1000*1000,
.callback = tim_timer_handler,
},
};
// Timer init
tim_init();
tim_config(HS_TIM0, &timer_cfg);
tim_start(HS_TIM0);
while(1);
}
