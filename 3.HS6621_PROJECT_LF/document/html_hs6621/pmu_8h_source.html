798 __STATIC_FORCEINLINE void pmu_cpu_reset(void)
799 {
800  // flag
801  register_set1(&HS_PMU->SW_STATUS, PMU_SW_STATUS_REBOOT_FROM_SOFT_RESET_MASK);
802 
803  // Use DMB to wait until all outstanding
804  // memory accesses are completed
805  __DMB();
806 
807 #if 0
808  register_set0(&HS_PMU->AHB_REMAP, PMU_CPU_SOFT_RESET_MASK);
809 #else
810  // Read back PRIGROUP and merge with SYSRESETREQ
811  SCB->AIRCR = 0x05FA0001 | (SCB->AIRCR & 0x700);
812 #endif
813 }
