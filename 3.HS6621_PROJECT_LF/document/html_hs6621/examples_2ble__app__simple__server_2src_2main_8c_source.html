165 static void peripheral_init(void)
166 {
167  // Init GPIO
168  gpio_open();
169 
170  // Wakeup Pin
171  gpio_set_direction(pin_wakeup_mask, GPIO_INPUT);
172  gpio_set_interrupt(pin_wakeup_mask, GPIO_BOTH_EDGE);
173  gpio_set_interrupt_callback(gpio_handler);
174 
175  // LEDS
176 #ifdef CONFIG_ASIC
177  gpio_write(pin_leds_mask, GPIO_HIGH);
178 #else
179  gpio_write(pin_leds_mask, GPIO_LOW);
180 #endif
181  gpio_set_direction(pin_leds_mask, GPIO_OUTPUT);
182 
183  // wakeup
184  pmu_wakeup_pin_set(pin_wakeup_mask, PMU_PIN_WAKEUP_LOW_LEVEL);
185  pmu_wakeup_pin_register_callback(wakeup_gpio_handler, wakeup_gpio_handler);
186 
187  // Init UART
188  uart_open(HS_UART0, DEBUG_UART_BAUDRATE, UART_FLOW_CTRL_DISABLED, NULL);
189 }

288 static void ble_stack_config(void)
289 {
290  // Disable WDT
291  wdt_enable(0);
292 
293  // Enable DCDC
294  pmu_dcdc_enable(true);
295 
296  // Power down xtal32k in deep sleep mode
297  pmu_32k_enable_in_deep_sleep(false);
298 
299  // Select 32k clock for stack
300  //pmu_xtal32k_change_param(15, 1);
301  pmu_select_32k(PMU_32K_SEL_RC);
302 
303  // xtal32m param
304  pmu_xtal32m_change_param(15);
305 
306  // ultra sleep mode enable
307  co_power_ultra_sleep_mode_enable(true);
308 
309  // Enable sleep, SWD will be closed.
310  co_power_sleep_enable(true);
311 }
