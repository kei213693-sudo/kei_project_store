92 uint8_t appm_whl_set_list(void)
93 {
94  uint8_t bond_num = 0; // when nvds modify, the bond num get from nvds
95  uint8_t nvds_len = NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST;
96  // BD Address of the peer device
97  struct gap_bdaddr peer_bd_addr[NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST];
98 
99  if (app_whl_env.enable == 0)
100  {
101  return 0;
102  }
103 
104  // get from nvds
105  if (app_sec_get_bond_status())
106  {
107  struct app_sec_conn_info_tag info;
108  uint8_t length = sizeof(struct app_sec_conn_info_tag);
109  uint8_t index = 0;
110 
111  // Get bond status from NVDS
112  for(index = 0; index <=nvds_len; index++)
113  {
114  if (nvds_get(NVDS_TAG_BLE_LINK_KEY_LAST-index, &length, (uint8_t *)&info) == NVDS_OK)
115  {
116  memcpy(&peer_bd_addr[bond_num], &info.id_addr, sizeof(struct gap_bdaddr));
117  bond_num++;
118  }
119  }
120  }
121 
122  if (bond_num == 0) return 0;
123 
124  struct gapm_list_set_wl_cmd* cmd = KE_MSG_ALLOC_DYN(GAPM_LIST_SET_CMD,
125  TASK_GAPM, TASK_APP,
126  gapm_list_set_wl_cmd, sizeof(struct gap_bdaddr)*bond_num);
127  cmd->operation = GAPM_SET_WL;
128 
129  cmd->size = bond_num;
130  for(uint8_t i=0;i<bond_num;i++)
131  {
132  memcpy(&cmd->wl_info[i], &peer_bd_addr[i], sizeof(struct gap_bdaddr));
133  }
134  ke_msg_send(cmd);
135  return bond_num;
136 }
