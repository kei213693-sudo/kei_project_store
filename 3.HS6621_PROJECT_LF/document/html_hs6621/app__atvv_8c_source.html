101 void app_atvv_recv_handler(struct atvv_server_write_ind const *param)
102 {
103  uint8_t data[32];
104 
105  memcpy(data, param->value, param->length);
106  log_debug_array_ex("[ATVV] atvv received data", data, param->length);
107 
108  switch(data[0])
109  {
110  case CMD_ATVV_GET_CAPS:
111  log_debug("[ATVV] CMD_ATVV_GET_CAPS\r\n");
112  _atvv_get_caps_respose(param->conidx);
113  break;
114 
115  case CMD_ATVV_MIC_OPEN:
116  log_debug("[ATVV] CMD_ATVV_MIC_OPEN\r\n");
117  _atvv_mic_open(param->conidx);
118  break;
119 
120  case CMD_ATVV_MIC_CLOSE:
121  log_debug("[ATVV] CMD_ATVV_MIC_CLOSE\r\n");
122  _atvv_mic_close(param->conidx);
123  break;
124 
125  case CMD_ATVV_SEARCH_END:
126  log_debug("[ATVV] CMD_ATVV_SEARCH_END\r\n");
127  _atvv_search_end();
128  break;
129 
130  default:
131  log_debug("[ATVV] unkown command\r\n");
132  break;
133  }
134 }

136 void app_atvv_notify_audio_frame(uint8_t *input, uint8_t len, int16_t prevsample, uint16_t previndex)
137 {
138  uint8_t buf[200];
139 
140  buf[0] = app_atvv_server_env.seq_num >> 8;
141  buf[1] = app_atvv_server_env.seq_num & 0x00FF;
142  buf[2] = 0x01;
143  buf[3] = (uint8_t)(prevsample >> 8);
144  buf[4] = prevsample & 0x00FF;
145  buf[5] = previndex & 0x00FF;
146 
147  memcpy(buf+6, input, len);
148  app_atvv_server_send_data(0, buf, len+6);
149 
150  app_atvv_server_env.seq_num++;
151 }
