92 uint8_t appm_privacy_set_list(void)
93 {
94  uint8_t bond_num = 0; // when nvds modify, the bond num get from nvds
95  uint8_t nvds_len = NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST;
96  struct gapc_irk irk[NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST];
97  uint8_t loc_irk[GAP_KEY_LEN];
98  struct gapm_list_set_ral_cmd* cmd;
99 
100  if (app_privacy_env.enable == 0)
101  {
102  return 0;
103  }
104 
105  // get from nvds
106  if (app_sec_get_bond_status())
107  {
108  struct app_sec_conn_info_tag info;
109  uint8_t length = sizeof(struct app_sec_conn_info_tag);
110  uint8_t index = 0;
111 
112  // Get bond status from NVDS
113  for(index = 0; index <=nvds_len; index++)
114  {
115  if (nvds_get(NVDS_TAG_BLE_LINK_KEY_LAST-index, &length, (uint8_t *)&info) == NVDS_OK)
116  {
117  memcpy(&irk[bond_num], &info.irk, sizeof(struct gapc_irk));
118  bond_num++;
119  }
120  }
121 
122  length = GAP_KEY_LEN;
123  nvds_get(NVDS_TAG_LOC_IRK, &length, loc_irk);
124  }
125 
126  if (bond_num == 0)
127  {
128  cmd = KE_MSG_ALLOC_DYN(GAPM_LIST_SET_CMD,
129  TASK_GAPM, TASK_APP,
130  gapm_list_set_ral_cmd, sizeof(struct gap_ral_dev_info));
131  cmd->operation = GAPM_SET_RAL;
132  cmd->size = 1;
133  memset(&cmd->ral_info[0], 0, sizeof(sizeof(struct gap_ral_dev_info)));
134  cmd->ral_info[0].priv_mode = 1;
135  }
136  else
137  {
138  cmd = KE_MSG_ALLOC_DYN(GAPM_LIST_SET_CMD,
139  TASK_GAPM, TASK_APP,
140  gapm_list_set_ral_cmd, sizeof(struct gap_ral_dev_info)*bond_num);
141  cmd->operation = GAPM_SET_RAL;
142  cmd->size = bond_num;
143 
144  for(uint8_t i=0;i<bond_num;i++)
145  {
146  memcpy(&cmd->ral_info[i].addr, &irk[i].addr, sizeof(struct gap_bdaddr));
147  cmd->ral_info[i].priv_mode = 1;
148  memcpy(cmd->ral_info[i].peer_irk, irk[i].irk.key, GAP_KEY_LEN);
149  memcpy(cmd->ral_info[i].local_irk, loc_irk, GAP_KEY_LEN);
150  }
151  }
152  ke_msg_send(cmd);
153  return bond_num;
154 }
