132 static void app_send_indicate( void )
133 {
134  sendSize = 0;
135  if( g_send_wechat.offset < g_send_wechat.len ){
136  if( g_send_wechat.len - g_send_wechat.offset > 20 ){
137  sendSize = 20;
138  }else{
139  sendSize = g_send_wechat.len - g_send_wechat.offset;
140  }
141  app_wechat_send_notify(&g_send_wechat.buf[g_send_wechat.offset], sendSize);
142  }else{
143  g_send_wechat.len = 0;
144  g_send_wechat.offset = 0;
145  memset(g_send_wechat.buf, 0, PKT_LENGTH);
146  }
147 }

203 static int app_wechat_write_req_handler(ke_msg_id_t const msgid,
204  struct wechat_write_ind *param,
205  ke_task_id_t const dest_id,
206  ke_task_id_t const src_id)
207 {
208  int chunk_size = 0;
209  if(ke_state_get(TASK_APP) == APPM_CONNECTED){
210  if(param->length <= 20){
211  if( g_recv_data.len == 0 ){
212  BpFixHead *fix_head = (BpFixHead *)param->write_buf;
213  g_recv_data.len = ntohs( fix_head->nLength );
214  g_recv_data.offset = 0;
215  g_recv_data.buf = ke_malloc(g_recv_data.len, KE_MEM_KE_MSG);
216  }
217  chunk_size = g_recv_data.len - g_recv_data.offset;
218  chunk_size = chunk_size < param->length ? chunk_size : param->length;
219  memcpy( g_recv_data.buf + g_recv_data.offset, param->write_buf, chunk_size );
220  g_recv_data.offset += chunk_size;
221  if( g_recv_data.len <= g_recv_data.offset ){
222  m_wechat_handler->m_data_consume_func( g_recv_data.buf, g_recv_data.len );
223  ke_free(g_recv_data.buf);
224  g_recv_data.buf = NULL;
225  g_recv_data.len = 0;
226  g_recv_data.offset = 0;
227  }
228  }
229  }
230 
231  return (KE_MSG_CONSUMED);
232 }
