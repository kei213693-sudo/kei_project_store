72void app_sec_init()
73{
74 memset(&app_sec_env, 0, sizeof(app_sec_env));
75 /*------------------------------------------------------
76 * RETRIEVE BOND STATUS
77 *------------------------------------------------------*/
78 #if (NVDS_SUPPORT)
79 struct app_sec_conn_info_tag info;
80 nvds_tag_len_t length = sizeof(struct app_sec_conn_info_tag);
81 uint8_t index = 0;
82 uint8_t nvds_len = NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST;
83
84 app_sec_env.bonded = false;
85
86 // Get bond status from NVDS
87 for(index =0; index <= nvds_len; index++)
88 {
89 length = sizeof(struct app_sec_conn_info_tag);
90 memset(&info, 0, length);
91 if (nvds_get(NVDS_TAG_BLE_LINK_KEY_FIRST+index, &length, (uint8_t *)&info) == NVDS_OK)
92 {
93 app_sec_env.bonded = true;
94 break;
95 }
96 }
97 #endif //(NVDS_SUPPORT)
98}

253static int gapc_bond_req_ind_handler(ke_msg_id_t const msgid,
254 struct gapc_bond_req_ind const *param,
255 ke_task_id_t const dest_id,
256 ke_task_id_t const src_id)
257{
258 uint8_t conidx = KE_IDX_GET(src_id);
259 //log_debug("%s@%d, request=%x\n", __func__, __LINE__, param->request);
260
261 // Prepare the GAPC_BOND_CFM message
262 struct gapc_bond_cfm *cfm = KE_MSG_ALLOC(GAPC_BOND_CFM,
263 src_id, TASK_APP,
264 gapc_bond_cfm);
265
266 switch (param->request)
267 {
268 case (GAPC_PAIRING_REQ):
269 {
270 cfm->request = GAPC_PAIRING_RSP;
271
272 #ifndef BLE_APP_AM0
273 cfm->accept = false;
274
275 // Check if we are already bonded (Only one bonded connection is supported)
276 if (!app_sec_env.con_bonded[conidx])
277 #endif // !BLE_APP_AM0
278 {
279 cfm->accept = true;
280
281 #if (BLE_APP_HID || BLE_APP_HT)
282 // Pairing Features
283 cfm->data.pairing_feat.auth = GAP_AUTH_REQ_MITM_BOND;
284 #elif defined(BLE_APP_AM0)
285 #if (BLE_APP_SEC_CON == 1)
286 if (param->data.auth_req & GAP_AUTH_SEC_CON)
287 {
288 cfm->data.pairing_feat.auth = GAP_AUTH_REQ_SEC_CON_BOND;
289 app_sec_env.sec_con_enabled[conidx] = true;
290 }
291 else
292 {
293 cfm->data.pairing_feat.auth = GAP_AUTH_REQ_NO_MITM_BOND;
294 app_sec_env.sec_con_enabled[conidx] = false;
295 }
296 #else // !(BLE_APP_SEC_CON)
297 cfm->data.pairing_feat.auth = GAP_AUTH_REQ_NO_MITM_BOND;
298 app_sec_env.sec_con_enabled[conidx] = false;
299 #endif // (BLE_APP_SEC_CON)
300 #else
301 cfm->data.pairing_feat.auth = GAP_AUTH_REQ_NO_MITM_BOND;
302 #endif //(BLE_APP_HID || BLE_APP_HT)
303
304 #if (BLE_APP_HT)
305 cfm->data.pairing_feat.iocap = GAP_IO_CAP_DISPLAY_ONLY;
306 #else
307 cfm->data.pairing_feat.iocap = GAP_IO_CAP_NO_INPUT_NO_OUTPUT;
308 #endif //(BLE_APP_HT)
309
310 cfm->data.pairing_feat.key_size = 16;
311 cfm->data.pairing_feat.oob = GAP_OOB_AUTH_DATA_NOT_PRESENT;
312 cfm->data.pairing_feat.sec_req = GAP_NO_SEC;
313
314 #if (defined(BLE_APP_AM0))
315 cfm->data.pairing_feat.rkey_dist = GAP_KDIST_ENCKEY | GAP_KDIST_IDKEY;
316 #if BLE_APP_AM0
317 // No Keys required from the Initiator
318 cfm->data.pairing_feat.ikey_dist = 0;
319 #else
320 cfm->data.pairing_feat.ikey_dist = GAP_KDIST_ENCKEY | GAP_KDIST_IDKEY;
321 #endif
322 #else
323#if (BLE_APP_PRIVACY)
324 cfm->data.pairing_feat.ikey_dist = GAP_KDIST_ENCKEY | GAP_KDIST_IDKEY;
325 cfm->data.pairing_feat.rkey_dist = GAP_KDIST_ENCKEY | GAP_KDIST_IDKEY;
326#else
327 cfm->data.pairing_feat.ikey_dist = GAP_KDIST_IDKEY;
328 cfm->data.pairing_feat.rkey_dist = GAP_KDIST_ENCKEY;
329 cfm->data.pairing_feat.auth = GAP_AUTH_REQ_MITM_BOND;
330#endif
331 #endif // (defined(BLE_APP_AM0))
332 }
333 } break;
334
335 case (GAPC_LTK_EXCH):
336 {
337 // Counter
338 uint8_t counter;
339
340 cfm->accept = true;
341 cfm->request = GAPC_LTK_EXCH;
342
343 // Generate all the values
344 cfm->data.ltk.ediv = (uint16_t)co_rand_word();
345
346 for (counter = 0; counter < RAND_NB_LEN; counter++)
347 {
348 cfm->data.ltk.ltk.key[counter] = (uint8_t)co_rand_word();
349 cfm->data.ltk.randnb.nb[counter] = (uint8_t)co_rand_word();
350 }
351
352 for (counter = RAND_NB_LEN; counter < KEY_LEN; counter++)
353 {
354 cfm->data.ltk.ltk.key[counter] = (uint8_t)co_rand_word();
355 }
356
357 #if (0)// (NVDS_SUPPORT)
358 if (nvds_put(NVDS_TAG_LTK, NVDS_LEN_LTK,
359 (uint8_t *)&cfm->data.ltk) != NVDS_OK)
360 {
361 ASSERT_ERR(0);
362 }
363 #else
364 memcpy(&app_sec_env.info[conidx].ltk, &cfm->data.ltk, sizeof(struct gapc_ltk));
365 memcpy(&app_sec_env.info[conidx].ltk.randnb.nb[0], &cfm->data.ltk.randnb.nb[0], RAND_NB_LEN);
366 app_sec_env.info[conidx].ltk.ediv = cfm->data.ltk.ediv;
367 #endif // #if (NVDS_SUPPORT)
368 } break;
369
370
371 case (GAPC_IRK_EXCH):
372 {
373 #if (NVDS_SUPPORT)
374 nvds_tag_len_t addr_len = BD_ADDR_LEN;
375 #endif //(NVDS_SUPPORT)
376
377 cfm->accept = true;
378 cfm->request = GAPC_IRK_EXCH;
379
380 // Load IRK
381 memcpy(cfm->data.irk.irk.key, app_env.loc_irk, KEY_LEN);
382
383 #if (NVDS_SUPPORT)
384 if (nvds_get(NVDS_TAG_BD_ADDRESS, &addr_len, cfm->data.irk.addr.addr.addr) != NVDS_OK)
385 #endif //(NVDS_SUPPORT)
386 {
387 //ASSERT_ERR(0);
388 memcpy(cfm->data.irk.addr.addr.addr, &co_default_bdaddr,sizeof(co_default_bdaddr));
389 }
390 // load device address
391 cfm->data.irk.addr.addr_type = (cfm->data.irk.addr.addr.addr[5] & 0xC0) ? ADDR_RAND : ADDR_PUBLIC;
392 } break;
393
394
395 //#if (BLE_APP_HT)
396 case (GAPC_TK_EXCH):
397 {
398 // Generate a PIN Code- (Between 100000 and 999999)
399 uint32_t pin_code = (100000 + (co_rand_word()%900000));
400
401 cfm->accept = true;
402 cfm->request = GAPC_TK_EXCH;
403
404 // Set the TK value
405 memset(cfm->data.tk.key, 0, KEY_LEN);
406
407 cfm->data.tk.key[0] = (uint8_t)((pin_code & 0x000000FF) >> 0);
408 cfm->data.tk.key[1] = (uint8_t)((pin_code & 0x0000FF00) >> 8);
409 cfm->data.tk.key[2] = (uint8_t)((pin_code & 0x00FF0000) >> 16);
410 cfm->data.tk.key[3] = (uint8_t)((pin_code & 0xFF000000) >> 24);
411 } break;
412 //#endif //(BLE_APP_HT)
413 case (GAPC_NC_EXCH):
414 {
415 cfm->accept = true;
416 cfm->request = GAPC_NC_EXCH;
417 } break;
418
419 default:
420 {
421 ASSERT_ERR(0);
422 } break;
423 }
424
425 // Send the message
426 ke_msg_send(cfm);
427
428 return (KE_MSG_CONSUMED);
429}

431static int gapc_bond_ind_handler(ke_msg_id_t const msgid,
432 struct gapc_bond_ind const *param,
433 ke_task_id_t const dest_id,
434 ke_task_id_t const src_id)
435{
436 uint8_t conidx = KE_IDX_GET(src_id);
437
438 //log_debug("%s@%d, info=%x\n", __func__, __LINE__, param->info);
439
440 switch (param->info)
441 {
442 case (GAPC_PAIRING_SUCCEED):
443 {
444 // Update the bonding status in the environment
445 app_sec_env.bonded = true;
446 app_sec_env.con_bonded[conidx] = true;
447
448 // Update the bonding status in the environment
449 #if (0)//(PLF_NVDS)
450 if (nvds_put(NVDS_TAG_PERIPH_BONDED, NVDS_LEN_PERIPH_BONDED,
451 (uint8_t *)&app_sec_env.bonded) != NVDS_OK)
452 {
453 // An error has occurred during access to the NVDS
454 ASSERT_ERR(0);
455 }
456
457 // Set the BD Address of the peer device in NVDS
458 if (nvds_put(NVDS_TAG_PEER_BD_ADDRESS, NVDS_LEN_PEER_BD_ADDRESS,
459 (uint8_t *)gapc_get_bdaddr(conidx, SMPC_INFO_PEER)) != NVDS_OK)
460 {
461 // An error has occurred during access to the NVDS
462 ASSERT_ERR(0);
463 }
464 #else
465#if (NVDS_SUPPORT)
466 struct app_sec_conn_info_tag info;
467 nvds_tag_len_t length = sizeof(struct app_sec_conn_info_tag);
468 uint8_t index = 0;
469 uint8_t nvds_len = NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST;
470
471 memcpy(&app_sec_env.info[conidx].id_addr, (uint8_t *)gapc_get_bdaddr(conidx, SMPC_INFO_PEER), sizeof(struct gap_bdaddr));
472 //log_debug("rom,%x,\r\n", app_sec_env.info[conidx].ltk.ediv);
473 //log_debug_array_ex("nb,", &app_sec_env.info[conidx].ltk.randnb.nb[0], GAP_RAND_NB_LEN);
474 // Get bond status from NVDS
475 for(index =0; index <= nvds_len; index++)
476 {
477 length = sizeof(struct app_sec_conn_info_tag);
478 memset(&info, 0, length);
479 if (nvds_get(NVDS_TAG_BLE_LINK_KEY_FIRST+index, &length, (uint8_t *)&info) != NVDS_OK)
480 {
481 length = sizeof(struct app_sec_conn_info_tag);
482 nvds_put(NVDS_TAG_BLE_LINK_KEY_FIRST+index, length, (uint8_t*)&app_sec_env.info[conidx]);
483 break;
484 }
485 else
486 {
487 //log_debug("get,i=%d,%x,\r\n", index, info.ltk.ediv);
488 //log_debug_array_ex("nb,", &info.ltk.randnb.nb[0], GAP_RAND_NB_LEN);
489 }
490 }
491 if (index > nvds_len)
492 {
493 nvds_del(NVDS_TAG_BLE_LINK_KEY_FIRST);
494 length = sizeof(struct app_sec_conn_info_tag);
495 nvds_put(NVDS_TAG_BLE_LINK_KEY_FIRST, length, (uint8_t*)&app_sec_env.info[conidx]);
496 }
497#endif
498 #endif //(PLF_NVDS)
499
500 #if (BLE_APP_ANCSC)
501 ancsc_encrypted_cmd(KE_IDX_GET(src_id));
502 #endif
503 #if (BLE_APP_AMSC)
504 amsc_encrypted_cmd(KE_IDX_GET(src_id));
505 #endif
506 #if (BLE_APP_AM0)
507 app_am0_send_audio_init(KE_IDX_GET(src_id));
508 #endif //(BLE_APP_AM0)
509 #if (BLE_APP_WHITE_LIST==1 && BLE_APP_PRIVACY == 0)
510 appm_whl_add_device((uint8_t *)gapc_get_bdaddr(0, SMPC_INFO_PEER));
511 #endif
512 #if (BLE_APP_HID)
513 // Enable HID Service
514 app_hid_enable_prf(app_env.conidx);
515 #endif //(BLE_APP_HID)
516 } break;
517
518 case (GAPC_REPEATED_ATTEMPT):
519 {
520 appm_disconnect(KE_IDX_GET(src_id));
521 } break;
522
523 case (GAPC_IRK_EXCH):
524 {
525#if (NVDS_SUPPORT)
526 // Store peer identity in NVDS
527 //if (nvds_put(NVDS_TAG_PEER_IRK, NVDS_LEN_PEER_IRK, (uint8_t *)&param->data.irk) != NVDS_OK)
528 //{
529 // ASSERT_ERR(0);
530 //}
531 memcpy(&app_sec_env.info[conidx].irk, (uint8_t *)&param->data.irk, sizeof(struct gapc_irk));
532#endif // (NVDS_SUPPORT)
533 //log_debug_array_ex("irk ", (uint8_t *)&param->data.irk, NVDS_LEN_PEER_IRK);
534 //log_debug_array_ex("local irk ", app_env.loc_irk, KEY_LEN);
535#if (BLE_APP_WHITE_LIST)
536 appm_whl_add_device((uint8_t *)&param->data.irk.addr);
537#endif
538#if (BLE_APP_PRIVACY)
539 struct gap_ral_dev_info device;
540 memcpy(&device.addr, &param->data.irk.addr, sizeof(struct gap_bdaddr));
541 device.priv_mode = 1;
542 memcpy(device.peer_irk, param->data.irk.irk.key, GAP_KEY_LEN);
543 memcpy(device.local_irk, app_env.loc_irk, GAP_KEY_LEN);
544 appm_privacy_add_device((uint8_t*)&device);
545#endif
546 } break;
547
548 case (GAPC_PAIRING_FAILED):
549 {
550 //app_sec_send_security_req(0);
551 log_debug("pairing failed reason=%x\n", param->data.reason);
552 } break;
553
554 // In Secure Connections we get BOND_IND with SMPC calculated LTK
555 case (GAPC_LTK_EXCH) :
556 {
557 #if (BLE_APP_SEC_CON)
558 //del the if , when master pair, ,edir & nb = 0;
559 //if (app_sec_env.sec_con_enabled[conidx] == true)
560 {
561 #if (NVDS_SUPPORT)
562 // Store LTK in NVDS
563 //if (nvds_put(NVDS_TAG_LTK, NVDS_LEN_LTK,(uint8_t *)&param->data.ltk.ltk.key[0]) != NVDS_OK)
564 //{
565 // ASSERT_ERR(0);
566 //}
567 memcpy(&app_sec_env.info[conidx].ltk, (uint8_t *)&param->data.ltk, sizeof(struct gapc_ltk));
568 #endif // (NVDS_SUPPORT)
569 }
570 #endif // (BLE_APP_SEC_CON)
571 }
572 break;
573
574 default:
575 {
576 ASSERT_ERR(0);
577 } break;
578 }
579
580 return (KE_MSG_CONSUMED);
581}

595static int gapc_encrypt_req_ind_handler(ke_msg_id_t const msgid,
596 struct gapc_encrypt_req_ind const *param,
597 ke_task_id_t const dest_id,
598 ke_task_id_t const src_id)
599{
600 uint8_t conidx = KE_IDX_GET(src_id);
601
602 // Prepare the GAPC_ENCRYPT_CFM message
603 struct gapc_encrypt_cfm *cfm = KE_MSG_ALLOC(GAPC_ENCRYPT_CFM,
604 src_id, TASK_APP,
605 gapc_encrypt_cfm);
606
607 cfm->found = false;
608 if (app_sec_env.con_bonded[conidx])
609 {
610 if ((param->ediv == app_sec_env.info[conidx].ltk.ediv) &&
611 !memcmp(&param->rand_nb.nb[0], &app_sec_env.info[conidx].ltk.randnb.nb[0], sizeof(struct rand_nb)))
612 {
613 cfm->found = true;
614 cfm->key_size = 16;
615 memcpy(&cfm->ltk, &app_sec_env.info[conidx].ltk.ltk, sizeof(struct gap_sec_key));
616 }
617 }
618 else if (app_sec_env.bonded)
619 {
620#if (NVDS_SUPPORT)
621 struct app_sec_conn_info_tag info;
622 nvds_tag_len_t length = sizeof(struct app_sec_conn_info_tag);
623 uint8_t index = 0;
624 uint8_t nvds_len = NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST;
625
626 //app_sec_env.bonded = false;
627 //log_debug("ori,%x,\r\n", param->ediv);
628 //log_debug_array_ex("nb1,", &param->rand_nb.nb[0], GAP_RAND_NB_LEN);
629
630 // Get bond status from NVDS
631 for(index =0; index <= nvds_len; index++)
632 {
633 length = sizeof(struct app_sec_conn_info_tag);
634 memset(&info, 0, length);
635 if (nvds_get(NVDS_TAG_BLE_LINK_KEY_FIRST+index, &length, (uint8_t *)&info) == NVDS_OK)
636 {
637 if ((param->ediv == info.ltk.ediv) &&
638 !memcmp(&param->rand_nb.nb[0], &info.ltk.randnb.nb[0], sizeof(struct rand_nb)))
639 {

644 }
645 else
646 {
647 //log_debug("ok,i=%d,%x,\r\n", index, info.ltk.ediv);
648 //log_debug_array_ex("nb,", &info.ltk.randnb.nb[0], GAP_RAND_NB_LEN);
649 }
650 }
651 else
652 {
653 //log_debug("fail,i=%d,%x,\r\n", index, info.ltk.ediv);
654 //log_debug_array_ex("nb,", &info.ltk.randnb.nb[0], GAP_RAND_NB_LEN);
655 }
656 }
657
658#endif // #if (NVDS_SUPPORT)
659 }
660 /*
661 * else the peer device is not known, an error should trigger a new pairing procedure.
662 */

666
667 return (KE_MSG_CONSUMED);
668}
