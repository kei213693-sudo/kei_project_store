117STATIC_INLINE uint32_t co_time_diff(uint32_t me, uint32_t he)
118{
119#if (CO_MAX_SYS_TIME == 0xFFFFFFFF)
120 uint32_t diff1 = me - he;
121 uint32_t diff2 = he - me;
122#else
123 uint32_t diff1 = (me > he) ? (me - he) : (he - me);
124 uint32_t diff2 = CO_MAX_SYS_TIME - diff1 + 1;
125#endif
126
127 return MIN(diff1, diff2);
128}

168STATIC_INLINE uint32_t co_time_pasting_time(uint32_t time, uint32_t cur_time)
169{
170 uint32_t delay;
171
172#if (CO_MAX_SYS_TIME == 0xFFFFFFFF)
173 delay = time - cur_time;
174#else
175 if (time > cur_time)
176 delay = time - cur_time;
177 else
178 delay = CO_MAX_SYS_TIME - cur_time + time;
179#endif
180

192STATIC_INLINE bool co_time_delay_ms_past(uint32_t delay_ms, uint32_t *prev_past_time)
193{
194 bool pasted;
195 uint32_t cur_time = co_time();
196
197 if(*prev_past_time == CO_INVALID_SYS_TIME)
198 *prev_past_time = cur_time;
199
200 pasted = co_time_diff(cur_time, *prev_past_time) > CO_TIME_MS2SYS(delay_ms);
201 if(pasted)
202 *prev_past_time = cur_time;
203
204 return pasted;
205}
