120static void appm_adv_build_adv_data(struct app_adv_data_tag* adv_data)
121{
122 uint8_t pos = 0, len = 0;
123
124 // flags into gapm_adv
125 // appearance
126 if ((adv_data->include &APP_ADV_DATA_INCLUDE_APPEARANCE)==APP_ADV_DATA_INCLUDE_APPEARANCE) {
127 app_adv_env.adv_data[pos++] = 3;
128 app_adv_env.adv_data[pos++] = APP_ADV_DATATYPE_APPEARANCE;
129 app_adv_env.adv_data[pos++] = (uint8_t) ((adv_data->appearance & 0x00FF) >> 0);
130 app_adv_env.adv_data[pos++] = (uint8_t) ((adv_data->appearance & 0xFF00) >> 8);
131 }
132
133 // tx power level
134 if ((adv_data->include &APP_ADV_DATA_INCLUDE_TX_POWER_LEVEL)==APP_ADV_DATA_INCLUDE_TX_POWER_LEVEL) {
135 app_adv_env.adv_data[pos++] = 2;
136 app_adv_env.adv_data[pos++] = APP_ADV_DATATYPE_TX_POWER_LEVEL;
137 app_adv_env.adv_data[pos++] = (uint8_t) (adv_data->tx_power_level);
138 }
139
140 // uuid
141 if ((adv_data->include &APP_ADV_DATA_INCLUDE_UUID)==APP_ADV_DATA_INCLUDE_UUID) {
142 app_adv_env.adv_data[pos++] = adv_data->uuid.len+1;
143 app_adv_env.adv_data[pos++] = adv_data->uuid.type;
144 memcpy(&app_adv_env.adv_data[pos], adv_data->uuid.uuid, adv_data->uuid.len);
145 pos += adv_data->uuid.len;
146 }
147
148 // name
149 if ((adv_data->include &APP_ADV_DATA_INCLUDE_NAME)==APP_ADV_DATA_INCLUDE_NAME) {
150 uint8_t name[APP_ADV_MAX_LEN];
151 len = appm_get_dev_name(name);
152 if (pos+2+len > APP_ADV_MAX_LEN)
153 {
154 len = APP_ADV_MAX_LEN - pos - 2;
155 }
156 if (len > 0) {
157 app_adv_env.adv_data[pos++] = len + 1;
158 app_adv_env.adv_data[pos++] = adv_data->name_type;
159 memcpy(&app_adv_env.adv_data[pos], name, len);
160 }
161 pos += len;
162 }
163
164 // APP_ADV_DATATYPE_MANUFACTURER_SPECIFIC_DATA
165 if ((adv_data->include &APP_ADV_DATA_INCLUDE_SPEC_DATA)==APP_ADV_DATA_INCLUDE_SPEC_DATA) {
166 if (adv_data->manuf_specific_data.len + pos +2 > APP_ADV_MAX_LEN) {
167 len = APP_ADV_MAX_LEN - pos - 2;
168 }
169 else {
170 len = adv_data->manuf_specific_data.len;
171 }
172 if (len >0) {
173 app_adv_env.adv_data[pos++] = len + 1;
174 app_adv_env.adv_data[pos++] = APP_ADV_DATATYPE_MANUFACTURER_SPECIFIC_DATA;
175 memcpy(&app_adv_env.adv_data[pos], adv_data->manuf_specific_data.data, len);
176 }
177 pos += len;
178 }
179 app_adv_env.adv_data_len = pos;
180}

182static void appm_adv_start_advertising(void)
183{
184 //log_debug("%s@%d\n", __func__, __LINE__);
185 // Prepare the GAPM_ACTIVITY_START_CMD message
186 struct gapm_activity_start_cmd *p_cmd = KE_MSG_ALLOC(GAPM_ACTIVITY_START_CMD,
187 TASK_GAPM, TASK_APP,
188 gapm_activity_start_cmd);
189
190 p_cmd->operation = GAPM_START_ACTIVITY;
191 p_cmd->actv_idx = app_adv_env.actv_idx;
192 switch(app_adv_env.cur_mode)
193 {
194 case APP_ADV_MODE_DIRECTED:
195 p_cmd->u_param.adv_add_param.duration = app_adv_env.config.directed_timeout*100;
196 break;
197 case APP_ADV_MODE_FAST:
198 p_cmd->u_param.adv_add_param.duration = app_adv_env.config.fast_timeout*100;
199 break;
200 case APP_ADV_MODE_SLOW:
201 p_cmd->u_param.adv_add_param.duration = app_adv_env.config.slow_timeout*100;
202 break;
203 default:
204 break;
205 }
206 //log_debug("timeout = %d\n", p_cmd->u_param.adv_add_param.duration );
207 p_cmd->u_param.adv_add_param.max_adv_evt = 0;
208
209 // Send the message
210 ke_msg_send(p_cmd);
211
212 // Keep the current operation
213 app_adv_env.state = APP_ADV_STATE_STARTING;
214 // And the next expected operation code for the command completed event
215 app_adv_env.op = GAPM_START_ACTIVITY;
216}

293static void appm_adv_create_advertising(void)
294{
295 //log_debug("%s@%d\n", __func__, __LINE__);
296 if (app_adv_env.state == APP_ADV_STATE_IDLE)
297 {
298 // Prepare the GAPM_ACTIVITY_CREATE_CMD message
299 struct gapm_activity_create_adv_cmd *p_cmd = KE_MSG_ALLOC(GAPM_ACTIVITY_CREATE_CMD,
300 TASK_GAPM, TASK_APP,
301 gapm_activity_create_adv_cmd);
302
303 // Set operation code
304 p_cmd->operation = GAPM_CREATE_ADV_ACTIVITY;
305
306 // Fill the allocated kernel message
307 p_cmd->own_addr_type = GAPM_STATIC_ADDR;
308#if (BLE_APP_PRIVACY)
309 if (app_sec_get_bond_status()) {
310 p_cmd->own_addr_type = GAPM_GEN_RSLV_ADDR;
311 }
312#endif
313 p_cmd->adv_param.type = GAPM_ADV_TYPE_LEGACY;
314 p_cmd->adv_param.prop = GAPM_ADV_PROP_UNDIR_CONN_MASK;
315 p_cmd->adv_param.filter_pol = ADV_ALLOW_SCAN_ANY_CON_ANY;
316#if (BLE_APP_WHITE_LIST)
317 if (app_sec_get_bond_status()) {
318 p_cmd->adv_param.filter_pol = ADV_ALLOW_SCAN_WLST_CON_WLST;
319 }
320#endif
321 p_cmd->adv_param.prim_cfg.chnl_map = APP_ADV_CHMAP;
322 p_cmd->adv_param.prim_cfg.phy = GAP_PHY_LE_1MBPS;
323 #ifdef BLE_APP_ADV_DISC_MODE
324 p_cmd->adv_param.disc_mode = BLE_APP_ADV_DISC_MODE;
325 #else
326 p_cmd->adv_param.disc_mode = GAPM_ADV_MODE_GEN_DISC;
327 #endif
328 #ifdef BLE_APP_ADV_TX_POWER
329 p_cmd->adv_param.max_tx_pwr = BLE_APP_ADV_TX_POWER;
330 #else
331 p_cmd->adv_param.max_tx_pwr = 0;
332 #endif
333
334 if (app_adv_env.cur_mode == APP_ADV_MODE_DIRECTED) {
335 if(app_adv_env.config.directed_enabled == false) {
336 app_adv_env.cur_mode = APP_ADV_MODE_FAST;
337 }
338 else if (app_sec_get_bond_status() == false) {
339 app_adv_env.cur_mode = APP_ADV_MODE_FAST;
340 }
341 }
342 if (app_adv_env.cur_mode == APP_ADV_MODE_FAST && app_adv_env.config.fast_enabled == false)
343 {
344 app_adv_env.cur_mode = APP_ADV_MODE_SLOW;
345 }
346 app_adv_env.next_mode = app_adv_env.cur_mode;
347 switch(app_adv_env.cur_mode)
348 {
349 case APP_ADV_MODE_DIRECTED:
350 /*
351 * If the peripheral is already bonded with a central device, use the direct advertising
352 * procedure (BD Address of the peer device is stored in NVDS.
353 */
354 if (app_sec_get_bond_status())
355 {
356 // BD Address of the peer device
357 struct gap_bdaddr peer_bd_addr;
358#if (NVDS_SUPPORT)
359 struct app_sec_conn_info_tag info;
360 nvds_tag_len_t length = sizeof(struct app_sec_conn_info_tag);
361 uint8_t index = 0;
362
363 uint8_t nvds_len = NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST;
364
365 // Get bond status from NVDS
366 for(index = 0; index <=nvds_len; index++)
367 {
368 if (nvds_get(NVDS_TAG_BLE_LINK_KEY_LAST-index, &length, (uint8_t *)&info) == NVDS_OK)
369 {
370#if 0//(BLE_APP_PRIVACY)
371 memcpy(&peer_bd_addr, &info.irk.addr, sizeof(struct gap_bdaddr));
372#else
373 memcpy(&peer_bd_addr, &info.id_addr, sizeof(struct gap_bdaddr));
374#endif
375 break;
376 }
377 }
378#else
379 memset(&peer_bd_addr, 0, sizeof(struct gap_bdaddr));
380#endif
381 // Set the DIRECT ADVERTISING mode
382 p_cmd->adv_param.disc_mode = GAPM_ADV_MODE_NON_DISC;
383 if (app_adv_env.config.directed_timeout == 0) {
384 p_cmd->adv_param.prop = GAPM_ADV_PROP_DIR_CONN_HDC_MASK;
385 }
386 else {
387 p_cmd->adv_param.prop = GAPM_ADV_PROP_DIR_CONN_MASK;
388 }
389 // Copy the BD address of the peer device and the type of address
390 memcpy(&p_cmd->adv_param.peer_addr, &peer_bd_addr, sizeof(struct gap_bdaddr));
391 }
392 p_cmd->adv_param.prim_cfg.adv_intv_min = APP_ADV_DIRECT_INTERVAL;
393 p_cmd->adv_param.prim_cfg.adv_intv_max = APP_ADV_DIRECT_INTERVAL;
394 break;
395 case APP_ADV_MODE_FAST:
396 p_cmd->adv_param.prim_cfg.adv_intv_min = app_adv_env.config.fast_interval;
397 p_cmd->adv_param.prim_cfg.adv_intv_max = app_adv_env.config.fast_interval;
398 break;
399 case APP_ADV_MODE_SLOW:
400 p_cmd->adv_param.prim_cfg.adv_intv_min = app_adv_env.config.slow_interval;
401 p_cmd->adv_param.prim_cfg.adv_intv_max = app_adv_env.config.slow_interval;
402 break;
403 default:
404 break;
405 }
406
407 // Send the message
408 ke_msg_send(p_cmd);
409
410 // Keep the current operation
411 app_adv_env.state = APP_ADV_STATE_CREATING;
412 // And the next expected operation code for the command completed event
413 app_adv_env.op = GAPM_CREATE_ADV_ACTIVITY;
414 }
415}

439static void appm_adv_fsm_next(void)
440{
441 //log_debug("%s@%d, mode=%x, state=%x, next event=%x\r\n", __func__, __LINE__,
442 // app_adv_env.cur_mode,
443 // app_adv_env.state,
444 // app_adv_env.next_event);
445 switch (app_adv_env.state)
446 {
447 case (APP_ADV_STATE_IDLE):
448 {
449 //log_debug("idle\n");
450 // Create advertising
451 appm_adv_create_advertising();
452 } break;
453
454 case (APP_ADV_STATE_CREATING):
455 {
456 //log_debug("creating\n");
457 if (app_adv_env.cur_mode == APP_ADV_MODE_DIRECTED)
458 {
459 // Start advertising activity
460 if (app_adv_env.next_event & APP_ADV_NEXT_EVENT_START)
461 {
462 appm_adv_start_advertising();
463 }
464 }
465 else
466 {
467 // Set advertising data
468 appm_adv_set_adv_data_cmd();
469 }
470 } break;
471
472 case (APP_ADV_STATE_SETTING_ADV_DATA):
473 {
474 //log_debug("setting adv data\n");
475 if (app_adv_env.res_data_len > 0 ) {
476 // Set scan response data
477 appm_adv_set_scan_rsp_data_cmd();
478 }
479 else if (app_adv_env.next_event & APP_ADV_NEXT_EVENT_START)
480 {
481 appm_adv_start_advertising();
482 }
483 } break;
484
485 case (APP_ADV_STATE_CREATED):
486 case (APP_ADV_STATE_SETTING_SCAN_RSP_DATA):
487 {
488 if (app_adv_env.state == APP_ADV_STATE_CREATED) {
489 //log_debug("created\n");
490 }
491 else {
492 //log_debug("setting scan res data\n");
493 }
494 // Start advertising activity
495 if (app_adv_env.next_event & APP_ADV_NEXT_EVENT_START)
496 {
497 appm_adv_start_advertising();
498 }
499 } break;
500
501 case (APP_ADV_STATE_STARTING):
502 {
503 //log_debug("starting\n");
504 #if (DISPLAY_SUPPORT)
505 // Update advertising state screen
506 app_display_set_adv(true);
507 #endif //(DISPLAY_SUPPORT)
508
509 // Go to started state
510 app_adv_env.state = APP_ADV_STATE_STARTED;
511 if (app_adv_env.next_event & APP_ADV_NEXT_EVENT_START)
512 {
513 app_adv_env.next_event &= (~APP_ADV_NEXT_EVENT_START);
514 }
515 else if ((app_adv_env.next_event & APP_ADV_NEXT_EVENT_STOP) || (app_adv_env.next_event & APP_ADV_NEXT_EVENT_DEL))
516 {
517 appm_adv_stop_advertising();
518 }
519 } break;
520
521 case (APP_ADV_STATE_STARTED):
522 {
523 //log_debug("started\n");
524
525 // Stop advertising activity
526 appm_adv_stop_advertising();
527 } break;
528
529 case (APP_ADV_STATE_STOPPING):
530 {
531 //log_debug("stopping\n");
532
533 #if (DISPLAY_SUPPORT)
534 // Update advertising state screen
535 app_adv_display_set_adv(false);
536 #endif //(DISPLAY_SUPPORT)
537
538 // Go created state
539 app_adv_env.state = APP_ADV_STATE_CREATED;
540 if(app_adv_env.cur_mode != app_adv_env.next_mode) {
541 appm_adv_delete_advertising();
542 }
543 else
544 {
545 if (app_adv_env.next_event & APP_ADV_NEXT_EVENT_STOP)
546 {
547 app_adv_env.next_event &= (~APP_ADV_NEXT_EVENT_STOP);
548 }
549
550 if (app_adv_env.next_event & APP_ADV_NEXT_EVENT_DEL)
551 {
552 app_adv_env.next_event &= (~APP_ADV_NEXT_EVENT_DEL);
553 appm_adv_delete_advertising();
554 }
555 else if (app_adv_env.next_event & APP_ADV_NEXT_EVENT_START)
556 {
557 app_adv_env.next_event &= (~APP_ADV_NEXT_EVENT_START);
558 appm_adv_start_advertising();
559 }
560 }
561 } break;
562
563 case (APP_ADV_STATE_DELETING):
564 {
565 //log_debug("deleting\n");
566 app_adv_env.state = APP_ADV_STATE_IDLE;
567 if(app_adv_env.cur_mode != app_adv_env.next_mode) {
568 app_adv_env.cur_mode = app_adv_env.next_mode;
569 if (app_adv_env.cur_mode != APP_ADV_MODE_IDLE) {
570 app_adv_env.next_event |= APP_ADV_NEXT_EVENT_START;
571 appm_adv_create_advertising();
572 }
573 else
574 {
575 if (app_adv_env.config.directed_enabled == true)
576 {
577 app_adv_env.cur_mode = APP_ADV_MODE_DIRECTED;
578 }
579 else if (app_adv_env.config.fast_enabled == true)
580 {
581 app_adv_env.cur_mode = APP_ADV_MODE_FAST;
582 }
583 else if (app_adv_env.config.slow_enabled == true)
584 {
585 app_adv_env.cur_mode = APP_ADV_MODE_SLOW;
586 }
587 if (app_adv_env.next_event & APP_ADV_NEXT_EVENT_START)
588 {
589 appm_adv_create_advertising();
590 }
591 }
592 }
593 else
594 {
595 if (app_adv_env.next_event & APP_ADV_NEXT_EVENT_START)
596 {
597 appm_adv_create_advertising();
598 }
599 }
600 }break;
601
602 default:
603 {
604 ASSERT_ERR(0);
605 } break;
606 }
607}

624void appm_adv_init(struct app_adv_data_tag* adv_data, struct app_adv_modes_config_tag *config)
625{
626 memset(&app_adv_env, 0, sizeof(struct app_adv_env_tag));
627 if (adv_data != NULL)
628 {
629 appm_adv_build_adv_data(adv_data);
630 }
631 if (config != NULL)
632 {
633 memcpy(&app_adv_env.config, config, sizeof(struct app_adv_modes_config_tag));
634
635 #if (BLE_APP_WHITE_LIST)
636 appm_whl_init(app_adv_env.config.whitelist_enabled);
637 #endif
638 #if (BLE_APP_PRIVACY)
639 appm_privacy_init(1);
640 #endif
641 if (config->fast_interval == 0)
642 {
643 app_adv_env.config.fast_enabled = false;
644 }
645 if (config->slow_interval == 0)
646 {
647 app_adv_env.config.slow_enabled = false;
648 }
649
650 if (app_adv_env.config.directed_enabled == true)
651 {
652 app_adv_env.cur_mode = APP_ADV_MODE_DIRECTED;
653 }
654 else if (app_adv_env.config.fast_enabled == true)
655 {
656 app_adv_env.cur_mode = APP_ADV_MODE_FAST;
657 }
658 else if (app_adv_env.config.slow_enabled == true)
659 {
660 app_adv_env.cur_mode = APP_ADV_MODE_SLOW;
661 }
662 else
663 {
664 ASSERT_ERR(0);
665 }
666 }
667}

704{
705 struct gapm_activity_stopped_ind *param = (struct gapm_activity_stopped_ind *)p_param;
706
707 //log_debug("%s@%d,%x\n", __func__, __LINE__, param->reason);
708
709 if (param->actv_type == GAPM_ACTV_TYPE_ADV) {
710
711 if (app_adv_env.state == APP_ADV_STATE_STARTED && app_adv_env.actv_idx == param->actv_idx)
712 {
713 // Act as if activity had been stopped by the application
714 app_adv_env.state = APP_ADV_STATE_STOPPING;
715 if (param->reason == GAP_ERR_TIMEOUT) {
716 app_adv_env.next_mode = app_adv_env.cur_mode+1;

719 {
720 app_adv_env.next_mode = APP_ADV_MODE_IDLE;
721 }
722 }
723 else if (param->reason == 0) // connect success
724 {
725 // adv mode >= 2, and mode == last mode. need del
726
727 uint8_t adv_type_num = 0;
728 uint8_t last_mode = APP_ADV_MODE_IDLE;
729 if (app_adv_env.config.directed_enabled) adv_type_num++;
730 if (app_adv_env.config.fast_enabled) {
731 adv_type_num++;
732 last_mode = APP_ADV_MODE_FAST;
733 }
734 if (app_adv_env.config.slow_enabled) {
735 adv_type_num++;
736 last_mode = APP_ADV_MODE_SLOW;
737 }
738 if (adv_type_num >=2 && app_adv_env.cur_mode == last_mode) {
739 app_adv_env.next_mode = app_adv_env.cur_mode+1;
740 if(app_adv_env.next_mode > APP_ADV_MODE_SLOW)
741 {
742 app_adv_env.next_mode = APP_ADV_MODE_IDLE;
743 }
744 }
