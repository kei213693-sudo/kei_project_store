55static uint8_t app_get_handler(const struct app_subtask_handlers *handler_list_desc,
56 ke_msg_id_t msgid,
57 void *param,
58 ke_task_id_t src_id)
59{
60 // Counter
61 uint8_t counter;
62
63 // Get the message handler function by parsing the message table
64 for (counter = handler_list_desc->msg_cnt; 0 < counter; counter--)
65 {
66 struct ke_msg_handler *handler
67 = (struct ke_msg_handler*)(handler_list_desc->p_msg_handler_tab + counter - 1);
68
69 if ((handler->id == msgid) ||
70 (handler->id == KE_MSG_DEFAULT_HANDLER))
71 {
72 // If handler is NULL, message should not have been received in this state
73 ASSERT_ERR(handler->func);
74
75 return (uint8_t)(handler->func(msgid, param, TASK_APP, src_id));
76 }
77 }
78
79 // If we are here no handler has been found, drop the message
80 return (KE_MSG_CONSUMED);
81}

183static int gapm_cmp_evt_handler(ke_msg_id_t const msgid,
184 struct gapm_cmp_evt const *param,
185 ke_task_id_t const dest_id,
186 ke_task_id_t const src_id)
187{
188 #if ((NVDS_SUPPORT) && (BLE_APP_SEC))
189 nvds_tag_len_t key_len = KEY_LEN;
190 #endif //(NVDS_SUPPORT)
191
192 switch(param->operation)
193 {
194 // Reset completed
195 case (GAPM_RESET):
196 {
197
198 if(param->status == GAP_ERR_NO_ERROR)
199 {
200 #if (NVDS_SUPPORT)
201 nvds_tag_len_t len=6;
202 #endif //(NVDS_SUPPORT)
203 #if (BLE_APP_HID)
204 app_hid_start_mouse();
205 #endif //(BLE_APP_HID)
206
207 // Set Device configuration
208 struct gapm_set_dev_config_cmd* cmd = KE_MSG_ALLOC(GAPM_SET_DEV_CONFIG_CMD,
209 TASK_GAPM, TASK_APP,
210 gapm_set_dev_config_cmd);
211 // Set the operation
212 cmd->operation = GAPM_SET_DEV_CONFIG;
213 // Set the device role - Peripheral
214 cmd->role = GAP_ROLE_PERIPHERAL;
215
216 #if (BLE_APP_SEC_CON)
217 // The Max MTU is increased to support the Public Key exchange
218 // HOWEVER, with secure connections enabled you cannot sniff the
219 // LEAP and LEAS protocols
220 cmd->max_mtu = 160;
221 cmd->pairing_mode = GAPM_PAIRING_SEC_CON | GAPM_PAIRING_LEGACY;
222 #else // !(BLE_APP_SEC_CON)
223 // Do not support secure connections
224 cmd->pairing_mode = GAPM_PAIRING_LEGACY;
225 #endif //(BLE_APP_SEC_CON)
226
227 // Set Data length parameters
228 cmd->sugg_max_tx_octets = BLE_MIN_OCTETS;
229 cmd->sugg_max_tx_time = BLE_MIN_TIME;
231 //cmd->tx_pref_phy = GAP_PHY_LE_2MBPS;
233 //cmd->rx_pref_phy = GAP_PHY_LE_2MBPS;
234 // Maximal MTU/MPS acceptable for device
235
236 #if (BLE_APP_HID)
237 // Enable Slave Preferred Connection Parameters present
238 cmd->att_cfg = GAPM_MASK_ATT_SLV_PREF_CON_PAR_EN;
239 #endif //(BLE_APP_HID)
240
241 // Host privacy enabled by default
242 cmd->privacy_cfg = 0;
243
244 #if (NVDS_SUPPORT)
245 if (nvds_get(NVDS_TAG_BD_ADDRESS, &len, &cmd->addr.addr[0]) != NVDS_OK)
246 {
247 memcpy(cmd->addr.addr, &co_default_bdaddr,sizeof(co_default_bdaddr));
248 }
249 #endif //(NVDS_SUPPORT)
250
251 // Host privacy enabled by default
252 #if (BLE_APP_PRIVACY)
253 cmd->privacy_cfg |= GAPM_PRIV_CFG_PRIV_EN_BIT;
254 cmd->renew_dur = 0x0384; // 15min
255 #endif
256
257 #if (BLE_APP_AM0)
258 cmd->audio_cfg = GAPM_MASK_AUDIO_AM0_SUP;
259 cmd->att_cfg |= GAPM_MASK_ATT_SVC_CHG_EN;
260 #endif //(BLE_APP_AM0)
261
262
263 #if ((NVDS_SUPPORT) && (BLE_APP_SEC))
264
265 if ((app_sec_get_bond_status()==true) &&
266 (nvds_get(NVDS_TAG_LOC_IRK, &key_len, app_env.loc_irk) == NVDS_OK))
267 {
268 memcpy(cmd->irk.key, app_env.loc_irk, 16);
269 }
270 else
271 #endif //(NVDS_SUPPORT)
272 {
273 memset((void *)&cmd->irk.key[0], 0x00, KEY_LEN);
274 }
275 // Send message
276 ke_msg_send(cmd);
277 }
278 else
279 {
280 ASSERT_ERR(0);
281 }
282 }
283 break;
284
285 case (GAPM_PROFILE_TASK_ADD):
286 {
287 // Add the next requested service
288 if (!appm_add_svc())
289 {
290 // Go to the ready state
291 ke_state_set(TASK_APP, APPM_READY);
292
293 // No more service to add, start advertising
294 //appm_adv_update_state(true);
295 appm_adv_start();
296 }
297 }
298 break;
299
300 // Device Configuration updated
301 #if (BLE_APP_WHITE_LIST)
302 case (GAPM_SET_DEV_CONFIG):
303 case (GAPM_GET_WLIST_SIZE):
304 case (GAPM_SET_WL):
305 {
306 if(appm_whl_cmp_evt_handler(msgid, param, dest_id, src_id) != KE_MSG_CONSUMED)
307 {
308 ke_state_t state = ke_state_get(dest_id);
309
310 if (state < APPM_READY) {
311 appm_add_svc();
312 }
313 }
314 }
315 break;
316 #else
317 case (GAPM_SET_DEV_CONFIG):
318 {
319 ASSERT_INFO(param->status == GAP_ERR_NO_ERROR, param->operation, param->status);
320
321 // Add the first required service in the database
322 // and wait for the PROFILE_ADDED_IND
323 appm_add_svc();
324 }
325 break;
326 #endif
327
328 #if (BLE_APP_PRIVACY)
329 case (GAPM_GET_RAL_SIZE):
330 case (GAPM_SET_RAL):
331 appm_privacy_cmp_evt_handler(msgid, param, dest_id, src_id);
332 break;
333 #endif
334
335 case (GAPM_CREATE_ADV_ACTIVITY):
336 case (GAPM_STOP_ACTIVITY):
337 case (GAPM_START_ACTIVITY):
338 case (GAPM_DELETE_ACTIVITY):
339 case (GAPM_SET_ADV_DATA):
340 case (GAPM_SET_SCAN_RSP_DATA):
341 case (GAPM_DELETE_ALL_ACTIVITIES) :
342 {
343 appm_adv_cmp_evt_handler((void*)param);
344 } break;
345
346 default:
347 {
348 // Drop the message
349 }
350 break;
351 }
352
353 return (KE_MSG_CONSUMED);
354}

456static int gapc_connection_req_ind_handler(ke_msg_id_t const msgid,
457 struct gapc_connection_req_ind const *param,
458 ke_task_id_t const dest_id,
459 ke_task_id_t const src_id)
460{
461 app_env.conidx = KE_IDX_GET(src_id);
462 #if(BLE_APP_SEC)
463 bool is_bond = app_sec_get_bond_status_by_addr(param->peer_addr);
464 #endif
465
466 log_debug("Device type(%d) conidx(%d) connected, ", param->peer_addr_type, app_env.conidx);
467 log_debug_array_ex("ADDR", &param->peer_addr, 6);
468 // Check if the received Connection Handle was valid
469 if (app_env.conidx != GAP_INVALID_CONIDX)
470 {
471 // Retrieve the connection info from the parameters
472 app_env.conhdl = param->conhdl;
473
474 // Send connection confirmation
475 struct gapc_connection_cfm *cfm = KE_MSG_ALLOC(GAPC_CONNECTION_CFM,
476 KE_BUILD_ID(TASK_GAPC, app_env.conidx), TASK_APP,
477 gapc_connection_cfm);
478
479 #if(BLE_APP_SEC)
480 cfm->auth = is_bond ? GAP_AUTH_REQ_NO_MITM_BOND : GAP_AUTH_REQ_NO_MITM_NO_BOND; // TODO [FBE] restore valid data
481 #else // !(BLE_APP_SEC)
482 cfm->auth = GAP_AUTH_REQ_NO_MITM_NO_BOND;
483 #endif // (BLE_APP_SEC)
484 // Send the message
485 ke_msg_send(cfm);
486
487 /*--------------------------------------------------------------
488 * ENABLE REQUIRED PROFILES
489 *--------------------------------------------------------------*/
490
491 #if (BLE_APP_BATT)
492 // Enable Battery Service
493 app_batt_enable_prf(app_env.conhdl);
494 #endif //(BLE_APP_BATT)
495
496 #if (BLE_APP_SIMPLE_SERVER)
497 // Enable SIMPLE_SERVER Service
498 app_simple_server_enable_prf(app_env.conhdl);
499 #endif //(BLE_APP_SIMPLE_SERVER)
500
501 // We are now in connected State
502 ke_state_set(dest_id, APPM_CONNECTED);
503
504
505 #if (BLE_APP_SEC)
506 if (is_bond)
507 {
508 // Ask for the peer device to either start encryption
509 app_sec_send_security_req(app_env.conidx);
510 }
511 #endif // (BLE_APP_SEC)
512 }
513 else
514 {
515 // No connection has been established, restart advertising
516 //appm_adv_update_state(true);
517 appm_adv_start();
518 }
519
520 return (KE_MSG_CONSUMED);
521}

632static int appm_msg_handler(ke_msg_id_t const msgid,
633 void *param,
634 ke_task_id_t const dest_id,
635 ke_task_id_t const src_id)
636{
637 // Retrieve identifier of the task from received message
638 ke_task_id_t src_task_id = MSG_T(msgid);
639 // Message policy
640 uint8_t msg_pol = KE_MSG_CONSUMED;
641
642 switch (src_task_id)
643 {
644 case (TASK_ID_GAPC):
645 {
646 #if (BLE_APP_SEC)
647 if ((msgid >= GAPC_BOND_CMD) &&
648 (msgid <= GAPC_SECURITY_IND))
649 {
650 // Call the Security Module
651 msg_pol = app_get_handler(&app_sec_handlers, msgid, param, src_id);
652 }
653 #endif //(BLE_APP_SEC)
654 // else drop the message
655 } break;
656
657 case (TASK_ID_GATTC):
658 {
659 // Service Changed - Drop
660 } break;
661
662 #if (BLE_APP_DIS)
663 case (TASK_ID_DISS):
664 {
665 // Call the Device Information Module
666 msg_pol = app_get_handler(&app_dis_handlers, msgid, param, src_id);
667 } break;
668 #endif //(BLE_APP_DIS)
669
670 #if (BLE_APP_SIMPLE_SERVER)
671 case (TASK_ID_SIMPLE_SERVER):
672 {
673 // Call the Audio Mode 0 Module
674 msg_pol = app_get_handler(&app_simple_server_handlers, msgid, param, src_id);
675 } break;
676
677 #endif //(BLE_APP_SIMPLE_SERVER)
678
679 default:
680 {
681 } break;
682 }
683
684 return (msg_pol);
685}
