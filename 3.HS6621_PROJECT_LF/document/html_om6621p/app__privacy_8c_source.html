92uint8_t appm_privacy_set_list(void)
93{
94 uint8_t bond_num = 0; // when nvds modify, the bond num get from nvds
95 uint8_t nvds_len = NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST;
96 struct gapc_irk irk[NVDS_TAG_BLE_LINK_KEY_LAST-NVDS_TAG_BLE_LINK_KEY_FIRST];
97 uint8_t loc_irk[GAP_KEY_LEN];
98 struct gapm_list_set_ral_cmd* cmd;
99
100 if (app_privacy_env.enable == 0)
101 {
102 return 0;
103 }
104
105 // get from nvds
106 if (app_sec_get_bond_status())
107 {
108 struct app_sec_conn_info_tag info;
109 nvds_tag_len_t length = sizeof(struct app_sec_conn_info_tag);
110 uint8_t index = 0;
111
112 // Get bond status from NVDS
113 for(index = 0; index <=nvds_len; index++)
114 {
115 if (nvds_get(NVDS_TAG_BLE_LINK_KEY_LAST-index, &length, (uint8_t *)&info) == NVDS_OK)
116 {
117 memcpy(&irk[bond_num], &info.irk, sizeof(struct gapc_irk));
118 bond_num++;
119 }
120 }
121
122 length = GAP_KEY_LEN;
123 if(nvds_get(NVDS_TAG_LOC_IRK, &length, loc_irk) != NVDS_OK)
124 {
125 memset(loc_irk, 0x00, GAP_KEY_LEN);
126 }
127 }
128
129 if (bond_num == 0)
130 {
131 cmd = KE_MSG_ALLOC_DYN(GAPM_LIST_SET_CMD,
132 TASK_GAPM, TASK_APP,
133 gapm_list_set_ral_cmd, sizeof(struct gap_ral_dev_info));
134 cmd->operation = GAPM_SET_RAL;
135 cmd->size = 0;
136 memset(&cmd->ral_info[0], 0, sizeof(sizeof(struct gap_ral_dev_info)));
137 cmd->ral_info[0].priv_mode = 1;
138
139
140 }
141 else
142 {
143 cmd = KE_MSG_ALLOC_DYN(GAPM_LIST_SET_CMD,
144 TASK_GAPM, TASK_APP,
145 gapm_list_set_ral_cmd, sizeof(struct gap_ral_dev_info)*bond_num);
146 cmd->operation = GAPM_SET_RAL;
147 cmd->size = bond_num;
148
149 for(uint8_t i=0;i<bond_num;i++)
150 {
151 memcpy(&cmd->ral_info[i].addr, &irk[i].addr, sizeof(struct gap_bdaddr));
152 cmd->ral_info[i].priv_mode = 1;
153 memcpy(cmd->ral_info[i].peer_irk, irk[i].irk.key, GAP_KEY_LEN);
154 memcpy(cmd->ral_info[i].local_irk, loc_irk, GAP_KEY_LEN);
155 }
156 }
157 ke_msg_send(cmd);
158 return bond_num;
159}
