197__STATIC_INLINE void cpm_ana_clock_set(uint32_t ana_if_ahb_cfg, uint32_t ana_if_clk_cfg)
198{
199 // Close ANA if clock
200 if (ana_if_ahb_cfg != CPM_ANA_CLK_IGNORE) HS_PSO->ANA_IF_AHB_CFG = ana_if_ahb_cfg;
201 if (ana_if_clk_cfg != CPM_ANA_CLK_IGNORE) HS_PSO->ANA_IF_CLK_CFG = ana_if_clk_cfg;
202 // Wait
203 HS_PSO_XTAL32M_UPD_RDY();
204}
